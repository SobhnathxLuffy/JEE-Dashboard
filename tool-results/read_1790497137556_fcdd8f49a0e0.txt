     1→     1→"use client";
     2→     2→
     3→     3→// ─── CBT / PDF test player — NTA palette, countdown, per-question timing ────
     4→     4→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     5→     5→import { Button } from "@/components/ui/button";
     6→     6→import { Input } from "@/components/ui/input";
     7→     7→import {
     8→     8→  AlertDialog,
     9→     9→  AlertDialogAction,
    10→    10→  AlertDialogCancel,
    11→    11→  AlertDialogContent,
    12→    12→  AlertDialogDescription,
    13→    13→  AlertDialogFooter,
    14→    14→  AlertDialogHeader,
    15→    15→  AlertDialogTitle,
    16→    16→  AlertDialogTrigger,
    17→    17→} from "@/components/ui/alert-dialog";
    18→    18→import { Badge } from "@/components/ui/badge";
    19→    19→import { cn } from "@/lib/utils";
    20→    20→import { saveSession, useLive } from "@/lib/idb";
    21→    21→import {
    22→    22→  SUBJECT_SHORT,
    23→    23→  todayStr,
    24→    24→  uid,
    25→    25→  type ActiveSession,
    26→    26→  type Question,
    27→    27→  type ResponseRecord,
    28→    28→  type Subject,
    29→    29→  type TestRecord,
    30→    30→} from "@/lib/types";
    31→    31→
    32→    32→const LETTERS = ["A", "B", "C", "D"];
    33→    33→
    34→    34→type SlotInfo = {
    35→    35→  slot: string; // question id (cbt) or number-string (pdf)
    36→    36→  no: number; // 1-based position
    37→    37→  subject: Subject;
    38→    38→  label: string;
    39→    39→};
    40→    40→
    41→    41→export function PlayerView({
    42→    42→  session,
    43→    43→  pdfBlob,
    44→    44→  onPersist,
    45→    45→  onFinish,
    46→    46→}: {
    47→    47→  session: ActiveSession;
    48→    48→  pdfBlob: Blob | null;
    49→    49→  onPersist: (s: ActiveSession) => void;
    50→    50→  onFinish: (testId: string) => void;
    51→    51→}) {
    52→    52→  const questions = useLive("questions");
    53→    53→  const [now, setNow] = useState(Date.now());
    54→    54→  const [pdfPage, setPdfPage] = useState<HTMLCanvasElement | null>(null);
    55→    55→  const [pdfErr, setPdfErr] = useState<string | null>(null);
    56→    56→  const [confirmOpen, setConfirmOpen] = useState(false);
    57→    57→  const submittingRef = useRef(false);
    58→    58→  const pdfDocRef = useRef<Awaited<ReturnType<typeof import("pdfjs-dist").getDocument>["promise"]> | null>(null);
    59→    59→  const sessionRef = useRef(session);
    60→    60→  sessionRef.current = session;
    61→    61→
    62→    62→  const qMap = useMemo(
    63→    63→    () => new Map(questions.map((q) => [q.id, q] as const)),
    64→    64→    [questions]
    65→    65→  );
    66→    66→
    67→    67→  const slots: SlotInfo[] = useMemo(() => {
    68→    68→    if (session.mode === "cbt") {
    69→    69→      const out: SlotInfo[] = [];
    70→    70→      let no = 1;
    71→    71→      for (const id of session.question_ids) {
    72→    72→        const q = qMap.get(id);
    73→    73→        if (!q) continue;
    74→    74→        out.push({ slot: id, no: no++, subject: q.subject, label: SUBJECT_SHORT[q.subject] });
    75→    75→      }
    76→    76→      return out;
    77→    77→    }
    78→    78→    const total = session.pdf_meta?.total_questions ?? 0;
    79→    79→    const subj = session.pdf_meta?.subject ?? "Physics";
    80→    80→    return Array.from({ length: total }, (_, i) => ({
    81→    81→      slot: String(i + 1),
    82→    82→      no: i + 1,
    83→    83→      subject: subj,
    84→    84→      label: SUBJECT_SHORT[subj],
    85→    85→    }));
    86→    86→  }, [session, qMap]);
    87→    87→
    88→    88→  const currentSlot = slots[Math.min(session.current, slots.length - 1)];
    89→    89→
    90→    90→  // sections = subjects in order of appearance
    91→    91→  const sections = useMemo(() => {
    92→    92→    const seen: Subject[] = [];
    93→    93→    for (const s of slots) {
    94→    94→      if (!seen.includes(s.subject)) seen.push(s.subject);
    95→    95→    }
    96→    96→    return seen;
    97→    97→  }, [slots]);
    98→    98→
    99→    99→  const sectionRanges = useMemo(() => {
   100→   100→    return sections.map((subj) => {
   101→   101→      const idxs = slots.map((s, i) => (s.subject === subj ? i : -1)).filter((i) => i >= 0);
   102→   102→      return { subject: subj, start: idxs[0] ?? 0, end: idxs[idxs.length - 1] ?? 0 };
   103→   103→    });
   104→   104→  }, [sections, slots]);
   105→   105→
   106→   106→  // ── timer ──
   107→   107→  useEffect(() => {
   108→   108→    const t = setInterval(() => setNow(Date.now()), 500);
   109→   109→    return () => clearInterval(t);
   110→   110→  }, []);
   111→   111→
   112→   112→  const remaining = Math.max(
   113→   113→    0,
   114→   114→    session.duration_min * 60 - (now - session.started_at) / 1000
   115→   115→  );
   116→   116→  const mins = Math.floor(remaining / 60);
   117→   117→  const secs = Math.floor(remaining % 60);
   118→   118→
   119→   119→  // ── per-question time accounting ──
   120→   120→  const commitTime = useCallback(
   121→   121→    (s: ActiveSession, nextIndex: number): ActiveSession => {
   122→   122→      const elapsed = (Date.now() - s.q_entered_at) / 1000;
   123→   123→      const cur = s.question_ids[s.current] ?? String(s.current + 1);
   124→   124→      return {
   125→   125→        ...s,
   126→   126→        q_times: { ...s.q_times, [cur]: (s.q_times[cur] ?? 0) + elapsed },
   127→   127→        current: nextIndex,
   128→   128→        q_entered_at: Date.now(),
   129→   129→      };
   130→   130→    },
   131→   131→    []
   132→   132→  );
   133→   133→
   134→   134→  const persist = useCallback(
   135→   135→    (s: ActiveSession) => {
   136→   136→      onPersist(s);
   137→   137→      void saveSession(s);
   138→   138→    },
   139→   139→    [onPersist]
   140→   140→  );
   141→   141→
   142→   142→  function goTo(index: number) {
   143→   143→    if (index < 0 || index >= slots.length) return;
   144→   144→    persist(commitTime(sessionRef.current, index));
   145→   145→  }
   146→   146→
   147→   147→  function setAnswer(slot: string, value: number | string | null) {
   148→   148→    const s = sessionRef.current;
   149→   149→    persist({ ...s, answers: { ...s.answers, [slot]: value } });
   150→   150→  }
   151→   151→
   152→   152→  function markAndNext() {
   153→   153→    const s = sessionRef.current;
   154→   154→    const cur = s.question_ids[s.current] ?? String(s.current + 1);
   155→   155→    const marked = s.marked.includes(cur) ? s.marked : [...s.marked, cur];
   156→   156→    const next = Math.min(s.current + 1, slots.length - 1);
   157→   157→    persist({ ...commitTime({ ...s, marked }, next), marked });
   158→   158→  }
   159→   159→
   160→   160→  function saveAndNext() {
   161→   161→    const s = sessionRef.current;
   162→   162→    const next = Math.min(s.current + 1, slots.length - 1);
   163→   163→    persist(commitTime(s, next));
   164→   164→  }
   165→   165→
   166→   166→  function clearResponse() {
   167→   167→    const s = sessionRef.current;
   168→   168→    if (!currentSlot) return;
   169→   169→    persist({
   170→   170→      ...s,
   171→   171→      answers: { ...s.answers, [currentSlot.slot]: null },
   172→   172→      marked: s.marked.filter((m) => m !== currentSlot.slot),
   173→   173→    });
   174→   174→  }
   175→   175→
   176→   176→  // ── submit ──
   177→   177→  const submit = useCallback(async () => {
   178→   178→    if (submittingRef.current) return;
   179→   179→    submittingRef.current = true;
   180→   180→    const s = sessionRef.current;
   181→   181→    // finalize last question's time
   182→   182→    const elapsed = (Date.now() - s.q_entered_at) / 1000;
   183→   183→    const cur = s.question_ids[s.current] ?? String(s.current + 1);
   184→   184→    const qTimes = { ...s.q_times, [cur]: (s.q_times[cur] ?? 0) + elapsed };
   185→   185→
   186→   186→    const responses: ResponseRecord[] = [];
   187→   187→    let score = 0;
   188→   188→    const subjectScores: Record<Subject, number> = {
   189→   189→      Physics: 0,
   190→   190→      Chemistry: 0,
   191→   191→      Mathematics: 0,
   192→   192→    };
   193→   193→
   194→   194→    if (s.mode === "cbt") {
   195→   195→      for (const id of s.question_ids) {
   196→   196→        const q = qMap.get(id);
   197→   197→        if (!q) continue;
   198→   198→        const raw = s.answers[id];
   199→   199→        const selected = raw === undefined || raw === "" || raw === null ? null : raw;
   200→   200→        const attempted = selected !== null;
   201→   201→        let correct = false;
   202→   202→        if (attempted) {
   203→   203→          if (q.type === "MCQ") correct = selected === q.answer;
   204→   204→          else {
   205→   205→            const u = Number(selected);
   206→   206→            correct =
   207→   207→              !Number.isNaN(u) &&
   208→   208→              Math.abs(u - Number(q.answer)) <= (q.tolerance || 0) + 1e-9;
   209→   209→          }
   210→   210→        }
   211→   211→        const marks = attempted ? (correct ? 4 : -1) : 0;
   212→   212→        score += marks;
   213→   213→        subjectScores[q.subject] += marks;
   214→   214→        responses.push({
   215→   215→          id: uid(),
   216→   216→          test_id: s.test_id,
   217→   217→          question_id: id,
   218→   218→          selected,
   219→   219→          correct,
   220→   220→          attempted,
   221→   221→          time_spent: Math.round(qTimes[id] ?? 0),
   222→   222→          error_tag: null,
   223→   223→          subject: q.subject,
   224→   224→          chapter: q.chapter,
   225→   225→          question_snippet: q.question.slice(0, 240),
   226→   226→          correct_answer: q.answer,
   227→   227→          type: q.type,
   228→   228→          options: q.options,
   229→   229→        });
   230→   230→      }
   231→   231→      const max = s.question_ids.length * 4;
   232→   232→      const record: TestRecord = {
   233→   233→        id: s.test_id,
   234→   234→        date: todayStr(),
   235→   235→        created_at: Date.now(),
   236→   236→        name: s.name,
   237→   237→        source: "in-app",
   238→   238→        type: s.test_type,
   239→   239→        duration_min: s.duration_min,
   240→   240→        score,
   241→   241→        max_score: max,
   242→   242→        subject_scores: subjectScores,
   243→   243→        question_ids: s.question_ids,
   244→   244→      };
   245→   245→      await (await import("@/lib/idb")).bulkPut("responses", responses);
   246→   246→      await (await import("@/lib/idb")).put("tests", record);
   247→   247→      onFinish(s.test_id);
   248→   248→      return;
   249→   249→    }
   250→   250→
   251→   251→    // PDF mode
   252→   252→    const meta = s.pdf_meta!;
   253→   253→    const keyMap = new Map(meta.key.map((k) => [k.no, k.answer]));
   254→   254→    for (let no = 1; no <= meta.total_questions; no++) {
   255→   255→      const raw = s.answers[String(no)];
   256→   256→      const selected =
   257→   257→        raw === undefined || raw === "" || raw === null ? null : String(raw);
   258→   258→      const attempted = selected !== null;
   259→   259→      const ans = keyMap.get(no);
   260→   260→      let correct = false;
   261→   261→      if (attempted && ans !== undefined) {
   262→   262→        if (/^[A-D]$/.test(ans)) correct = selected!.toUpperCase() === ans;
   263→   263→        else {
   264→   264→          const u = Number(selected);
   265→   265→          const a = Number(ans);
   266→   266→          correct = !Number.isNaN(u) && Math.abs(u - a) <= 1e-9;
   267→   267→        }
   268→   268→      }
   269→   269→      const marks = attempted ? (correct ? 4 : -1) : 0;
   270→   270→      score += marks;
   271→   271→      subjectScores[meta.subject] += marks;
   272→   272→      responses.push({
   273→   273→        id: uid(),
   274→   274→        test_id: s.test_id,
   275→   275→        question_id: `pdf:${s.test_id}:${no}`,
   276→   276→        selected: attempted ? selected : null,
   277→   277→        correct,
   278→   278→        attempted,
   279→   279→        time_spent: Math.round(qTimes[String(no)] ?? 0),
   280→   280→        error_tag: null,
   281→   281→        subject: meta.subject,
   282→   282→        chapter: meta.chapter,
   283→   283→        question_snippet: `PDF Q${no} (page ${pageForQuestion(s, no)})`,
   284→   284→        correct_answer: ans ?? "?",
   285→   285→        type: /^[A-D]$/.test(String(ans)) ? "MCQ" : "numerical",
   286→   286→        options: /^[A-D]$/.test(String(ans)) ? ["A", "B", "C", "D"] : [],
   287→   287→      });
   288→   288→    }
   289→   289→    const record: TestRecord = {
   290→   290→      id: s.test_id,
   291→   291→      date: todayStr(),
   292→   292→      created_at: Date.now(),
   293→   293→      name: s.name,
   294→   294→      source: "PDF",
   295→   295→      type: "pdf",
   296→   296→      duration_min: s.duration_min,
   297→   297→      score,
   298→   298→      max_score: meta.total_questions * 4,
   299→   299→      subject_scores: subjectScores,
   300→   300→      pdf_meta: meta,
   301→   301→    };
   302→   302→    const idb = await import("@/lib/idb");
   303→   303→    await idb.bulkPut("responses", responses);
   304→   304→    await idb.put("tests", record);
   305→   305→    onFinish(s.test_id);
   306→   306→  }, [qMap, onFinish]);
   307→   307→
   308→   308→  // auto-submit when the clock hits zero
   309→   309→  useEffect(() => {
   310→   310→    if (remaining <= 0 && !submittingRef.current) {
   311→   311→      void submit();
   312→   312→    }
   313→   313→  }, [remaining, submit]);
   314→   314→
   315→   315→  // warn before leaving with a live test
   316→   316→  useEffect(() => {
   317→   317→    const h = (e: BeforeUnloadEvent) => {
   318→   318→      e.preventDefault();
   319→   319→    };
   320→   320→    window.addEventListener("beforeunload", h);
   321→   321→    return () => window.removeEventListener("beforeunload", h);
   322→   322→  }, []);
   323→   323→
   324→   324→  const curNo = currentSlot ? currentSlot.no : 1;
   325→   325→
   326→   326→  // ── pdf.js: load document once, render current page ──
   327→   327→  useEffect(() => {
   328→   328→    let cancelled = false;
   329→   329→    if (session.mode !== "pdf" || !pdfBlob) {
   330→   330→      setPdfErr(session.mode === "pdf" ? "PDF file missing — discard the test and re-import." : null);
   331→   331→      return;
   332→   332→    }
   333→   333→    (async () => {
   334→   334→      try {
   335→   335→        const pdfjs = await import("pdfjs-dist");
   336→   336→        pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   337→   337→        if (!pdfDocRef.current) {
   338→   338→          const buf = await pdfBlob.arrayBuffer();
   339→   339→          pdfDocRef.current = await pdfjs.getDocument({ data: buf }).promise;
   340→   340→        }
   341→   341→        if (cancelled || !pdfDocRef.current) return;
   342→   342→        const pageNo = pageForQuestion(sessionRef.current, curNo);
   343→   343→        const page = await pdfDocRef.current.getPage(Math.min(pageNo, pdfDocRef.current.numPages));
   344→   344→        if (cancelled) return;
   345→   345→        const scale = 1.6;
   346→   346→        const viewport = page.getViewport({ scale });
   347→   347→        const canvas = document.createElement("canvas");
   348→   348→        canvas.width = viewport.width;
   349→   349→        canvas.height = viewport.height;
   350→   350→        const ctx = canvas.getContext("2d");
   351→   351→        if (!ctx) return;
   352→   352→        await page.render({ canvasContext: ctx, viewport }).promise;
   353→   353→        if (!cancelled) {
   354→   354→          setPdfPage(canvas);
   355→   355→          setPdfErr(null);
   356→   356→        }
   357→   357→      } catch (e) {
   358→   358→        if (!cancelled) setPdfErr(`PDF render failed: ${String(e)}`);
   359→   359→      }
   360→   360→    })();
   361→   361→    return () => {
   362→   362→      cancelled = true;
   363→   363→    };
   364→   364→  }, [session.mode, pdfBlob, curNo, session.current]);
   365→   365→
   366→   366→  if (slots.length === 0) {
   367→   367→    return (
   368→   368→      <div className="text-center py-20">
   369→   369→        <p className="text-stone-500">This test has no questions available.</p>
   370→   370→        <Button variant="outline" className="mt-4" onClick={() => onFinish("")}>
   371→   371→          Back
   372→   372→        </Button>
   373→   373→      </div>
   374→   374→    );
   375→   375→  }
   376→   376→
   377→   377→  const curQ: Question | undefined =
   378→   378→    session.mode === "cbt" && currentSlot ? qMap.get(currentSlot.slot) : undefined;
   379→   379→  const selected = currentSlot ? session.answers[currentSlot.slot] : undefined;
   380→   380→  const isMarked = currentSlot ? session.marked.includes(currentSlot.slot) : false;
   381→   381→  const answeredCount = Object.values(session.answers).filter(
   382→   382→    (v) => v !== null && v !== undefined && v !== ""
   383→   383→  ).length;
   384→   384→
   385→   385→  // active section derived from current question's subject
   386→   386→  const activeSection = currentSlot?.subject ?? sections[0];
   387→   387→
   388→   388→  function sectionTabClick(subj: Subject) {
   389→   389→    const range = sectionRanges.find((r) => r.subject === subj);
   390→   390→    if (range) goTo(range.start);
   391→   391→  }
   392→   392→
   393→   393→  return (
   394→   394→    <div className="space-y-4">
   395→   395→      {/* header */}
   396→   396→      <div
   397→   397→        className={cn(
   398→   398→          "bg-white border rounded-xl px-4 py-3 flex items-center justify-between gap-3 flex-wrap",
   399→   399→          remaining < 120 ? "border-red-300" : "border-stone-200"
   400→   400→        )}
   401→   401→      >
   402→   402→        <div className="min-w-0">
   403→   403→          <div className="font-semibold text-stone-900 truncate">{session.name}</div>
   404→   404→          <div className="text-xs text-stone-400">
   405→   405→            {slots.length} questions · +4 / −1 / 0 · numericals included
   406→   406→          </div>
   407→   407→        </div>
   408→   408→        <div className="flex items-center gap-3">
   409→   409→          <div
   410→   410→            className={cn(
   411→   411→              "text-2xl font-black tabular-nums px-4 py-1.5 rounded-lg",
   412→   412→              remaining < 120 ? "bg-red-50 text-red-600" : "bg-stone-100 text-stone-800"
   413→   413→            )}
   414→   414→            aria-live="polite"
   415→   415→          >
   416→   416→            {String(mins).padStart(2, "0")}:{String(secs).padStart(2, "0")}
   417→   417→          </div>
   418→   418→          <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   419→   419→            <AlertDialogTrigger asChild>
   420→   420→              <Button className="bg-emerald-700 hover:bg-emerald-800">Submit</Button>
   421→   421→            </AlertDialogTrigger>
   422→   422→            <AlertDialogContent>
   423→   423→              <AlertDialogHeader>
   424→   424→                <AlertDialogTitle>Submit test?</AlertDialogTitle>
   425→   425→                <AlertDialogDescription>
   426→   426→                  {answeredCount} answered · {slots.length - answeredCount} unattempted. After
   427→   427→                  submit you tag every mistake — that&apos;s where the marks come back.
   428→   428→                </AlertDialogDescription>
   429→   429→              </AlertDialogHeader>
   430→   430→              <AlertDialogFooter>
   431→   431→                <AlertDialogCancel>Keep solving</AlertDialogCancel>
   432→   432→                <AlertDialogAction
   433→   433→                  className="bg-emerald-700 hover:bg-emerald-800"
   434→   434→                  onClick={() => void submit()}
   435→   435→                >
   436→   436→                  Submit &amp; analyze
   437→   437→                </AlertDialogAction>
   438→   438→              </AlertDialogFooter>
   439→   439→            </AlertDialogContent>
   440→   440→          </AlertDialog>
   441→   441→        </div>
   442→   442→      </div>
   443→   443→
   444→   444→      <div className="grid lg:grid-cols-4 gap-4">
   445→   445→        {/* main panel */}
   446→   446→        <div className="lg:col-span-3 bg-white border border-stone-200 rounded-xl p-4 min-h-[420px] flex flex-col">
   447→   447→          {session.mode === "pdf" ? (
   448→   448→            <div className="flex-1 flex flex-col">
   449→   449→              <div className="flex items-center justify-between mb-2">
   450→   450→                <Badge variant="outline" className="border-stone-300 text-stone-500">
   451→   451→                  PDF page {pageForQuestion(session, curNo)}
   452→   452→                </Badge>
   453→   453→                <span className="text-xs text-stone-400">
   454→   454→                  {pdfErr ? "PDF render failed — check the file" : "question paper on the right →"}
   455→   455→                </span>
   456→   456→              </div>
   457→   457→              <div className="border border-stone-200 rounded-lg overflow-auto max-h-[420px] bg-stone-100">
   458→   458→                {pdfErr ? (
   459→   459→                  <div className="p-6 text-sm text-red-500">{pdfErr}</div>
   460→   460→                ) : pdfPage ? (
   461→   461→                  <canvas
   462→   462→                    className="w-full h-auto block"
   463→   463→                    ref={(node) => {
   464→   464→                      if (node && pdfPage) {

   738→   738→function pageForQuestion(s: ActiveSession, no: number): number {
   739→   739→  const meta = s.pdf_meta;
   740→   740→  if (!meta) return 1;
   741→   741→  const pages = Math.max(1, meta.end_page - meta.start_page + 1);
   742→   742→  const perPage = Math.ceil(meta.total_questions / pages);
   743→   743→  return meta.start_page + Math.floor((no - 1) / perPage);
   744→   744→}

   746→   746→function pdfKeyIsLetter(s: ActiveSession, no: number): boolean {
   747→   747→  const ans = s.pdf_meta?.key.find((k) => k.no === no)?.answer;
   748→   748→  return ans !== undefined && /^[A-D]$/.test(ans);
