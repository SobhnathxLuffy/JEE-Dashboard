    37→/**
    38→ * Answer-key text parser — tolerates the common dirty shapes:
    39→ * "1. A", "1A", "1) b", "7. B or C", "7 B/C", "9. bonus", "1. 42", "1. -12.5".
    40→ * Key numbers are paper numbering (whatever the sheet says).
    41→ */
    42→function parseAnswerKey(text: string): PdfKeyEntry[] {
    43→  const out: PdfKeyEntry[] = [];
    44→  const re =
    45→    /(\d{1,3})\s*[.):\-]?\s*(bonus|(?:[A-Da-d]\s*(?:\/|,|[oO][rR]\b)\s*)+[A-Da-d]|[A-Da-d])(?![A-Za-z0-9/])|(\d{1,3})\s*[.):\-]\s*(-?\d+(?:\.\d+)?)/g;
    46→  let m: RegExpExecArray | null;
    47→  while ((m = re.exec(text)) !== null) {
    48→    if (m[1] !== undefined && m[2] !== undefined) {
    49→      const v = m[2].trim();
    50→      if (/^bonus$/i.test(v)) {
    51→        out.push({ no: Number(m[1]), answer: "", bonus: true });
    52→      } else if (v.length > 1) {
    53→        const letters = [
    54→          ...new Set(
    55→            v
    56→              .split(/\s*(?:\/|,|[oO][rR]\b)\s*/)
    57→              .filter(Boolean)
    58→              .map((p) => p.toUpperCase())
    59→          ),
    60→        ].filter((p) => /^[A-D]$/.test(p));
    61→        if (letters.length > 1) {
    62→          out.push({ no: Number(m[1]), answer: letters[0], answers: letters });
    63→        } else if (letters.length === 1) {
    64→          out.push({ no: Number(m[1]), answer: letters[0] });
    65→        }
    66→      } else {
    67→        out.push({ no: Number(m[1]), answer: v.toUpperCase() });
    68→      }
    69→    } else if (m[3] !== undefined && m[4] !== undefined) {
    70→      out.push({ no: Number(m[3]), answer: m[4] });
    71→    }
    72→  }
    73→  // dedupe by question number (last wins)
    74→  const map = new Map<number, PdfKeyEntry>();
    75→  for (const k of out) map.set(k.no, k);
    76→  return [...map.values()].sort((a, b) => a.no - b.no);
    77→}

    79→type CellResult =
    80→  | { kind: "empty" }
    81→  | { kind: "entry"; entry: PdfKeyEntry }
    82→  | { kind: "invalid" };

    84→/** Interpret one key-grid cell (raw text) — option-number mapping applied here. */
    85→function parseCell(raw: string, optionNums: boolean): CellResult {
    86→  const t = raw.trim();
    87→  if (t === "") return { kind: "empty" };
    88→  if (/^bonus$/i.test(t)) {
    89→    return { kind: "entry", entry: { no: 0, answer: "", bonus: true } };
    90→  }
    91→  const parts = t
    92→    .split(/\s*(?:\/|,|[oO][rR]\b)\s*/)
    93→    .filter(Boolean);
    94→  if (parts.length > 1) {
    95→    const letters = [
    96→      ...new Set(parts.map((p) => p.toUpperCase())),
    97→    ].filter((p) => /^[A-D]$/.test(p));
    98→    if (letters.length > 1 && letters.length === parts.length) {
    99→      return { kind: "entry", entry: { no: 0, answer: letters[0], answers: letters } };
   100→    }
   101→    return { kind: "invalid" };
   102→  }
   103→  if (/^[A-Da-d]$/.test(t)) {
   104→    return { kind: "entry", entry: { no: 0, answer: t.toUpperCase() } };
   105→  }
   106→  if (optionNums && /^[1-4]$/.test(t)) {
   107→    return { kind: "entry", entry: { no: 0, answer: LETTERS[Number(t) - 1] } };
   108→  }
   109→  if (NUM_RE.test(t)) {
   110→    return { kind: "entry", entry: { no: 0, answer: t } };
   111→  }
   112→  return { kind: "invalid" };
   113→}

   264→  async function onFile(file: File) {
   265→    if (file.type !== "application/pdf" && !file.name.toLowerCase().endsWith(".pdf")) {
   266→      toast.error("That's not a PDF");
   267→      return;
   268→    }
   269→    setLoading(true);
   270→    try {
   271→      const pdfjs = await import("pdfjs-dist");
   272→      pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   273→      const buf = await file.arrayBuffer();
   274→      const doc = await pdfjs.getDocument({ data: buf.slice(0) }).promise;
   275→      setNumPages(doc.numPages);
   276→      doc.destroy();
   277→      setBlob(file);
   278→      setFileName(file.name);
   279→      setEndPage(String(Math.min(5, doc.numPages)));
   280→      if (!name) setName(`PDF test — ${file.name.replace(/\.pdf$/i, "").slice(0, 40)}`);
   281→      // persist immediately — a refresh before Start must not lose the file
   282→      await kvSet("pdf-blob", file);
   283→      toast.success(`Loaded ${doc.numPages} pages`);
   284→    } catch (e) {
   285→      toast.error(`Could not read PDF: ${String(e)}`);
   286→    } finally {
   287→      setLoading(false);
   288→    }
   289→  }

   291→  // ── A4: digital key-PDF → text layer → same parser → grid + textarea ──
   292→  async function onKeyFile(file: File) {
   293→    if (file.type !== "application/pdf" && !file.name.toLowerCase().endsWith(".pdf")) {
   294→      toast.error("That's not a PDF");
   295→      return;
   296→    }
   297→    setKeyLoading(true);
   298→    try {
   299→      const pdfjs = await import("pdfjs-dist");
   300→      pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.mjs";
   301→      const buf = await file.arrayBuffer();
   302→      const doc = await pdfjs.getDocument({ data: buf.slice(0) }).promise;
   303→      let text = "";
   304→      for (let p = 1; p <= doc.numPages; p++) {
   305→        const page = await doc.getPage(p);
   306→        const tc = await page.getTextContent();
   307→        text += tc.items.map((it) => ("str" in it ? it.str : "")).join(" ") + "\n";
   308→        if (text.length > 400_000) break; // safety cap
   309→      }
   310→      doc.destroy();
   311→      const parsed = parseAnswerKey(text.slice(0, 400_000));
   312→      if (parsed.length < 2) {
   313→        toast.error("No readable text — this key looks scanned. Paste it instead.");
   314→        return;
   315→      }
   316→      setKeyFileName(file.name);
   317→      setKeyText(text.slice(0, 400_000)); // fills grid via the paste effect
   318→      toast.success(`${parsed.length} keys extracted from key PDF`);
   319→    } catch (e) {
   320→      toast.error(`Could not read key PDF: ${String(e)}`);
   321→    } finally {
   322→      setKeyLoading(false);
   323→    }
   324→  }

   326→  async function startTest() {
   327→    if (!blob) {
   328→      toast.error("Upload a PDF first");
   329→      return;
   330→    }
   331→    const tol = Math.max(0, Number(tolerance) || 0);
   332→    let meta: NonNullable<ActiveSession["pdf_meta"]>;
   333→    let subjectOrder: Subject[];
   334→
   335→    if (mode === "full") {
   336→      const secs: PdfSection[] = [];
   337→      for (const s of SUBJECTS) {
   338→        const r = rows[s];
   339→        const fq = Number(r.first_q);
   340→        const lq = Number(r.last_q);
   341→        const spg = Number(r.start_page);
   342→        const epg = Number(r.end_page);
   343→        if (!Number.isFinite(fq) || !Number.isFinite(lq) || fq < 1 || lq < fq) {
   344→          toast.error(`${s}: check first / last question numbers`);
   345→          return;
   346→        }
   347→        if (!Number.isFinite(spg) || !Number.isFinite(epg) || spg < 1 || epg < spg) {
   348→          toast.error(`${s}: check start / end pages`);
   349→          return;
   350→        }
   351→        secs.push({
   352→          subject: s,
   353→          chapter: r.chapter || `${s} questions`,
   354→          first_q: fq,
   355→          last_q: lq,
   356→          start_page: spg,
   357→          end_page: epg,
   358→        });
   359→      }
   360→      secs.sort((a, b) => a.first_q - b.first_q);
   361→      for (let i = 1; i < secs.length; i++) {
   362→        if (secs[i].first_q <= secs[i - 1].last_q) {
   363→          toast.error("Sections overlap in question numbers — fix the section rows");
   364→          return;
   365→        }
   366→      }
   367→      meta = {
   368→        subject: secs[0].subject,
   369→        chapter: "Full paper",
   370→        start_page: Math.min(...secs.map((s) => s.start_page)),
   371→        end_page: Math.max(...secs.map((s) => s.end_page)),
   372→        total_questions: secs[secs.length - 1].last_q - secs[0].first_q + 1,
   373→        key: keyLater ? [] : gridKey,
   374→        key_later: keyLater,
   375→        sections: secs,
   376→        first_q: secs[0].first_q,
   377→        tolerance: tol,
   378→      };
   379→      subjectOrder = [...SUBJECTS];
   380→    } else {
   381→      const sp = Number(startPage) || 1;
   382→      const ep = Number(endPage) || sp;
   383→      const tq = Number(totalQ) || 0;
   384→      const fq = Number(firstQ) || 1;
   385→      if (tq <= 0) {
   386→        toast.error("Total questions must be ≥ 1");
   387→        return;
   388→      }
   389→      if (fq < 1) {
   390→        toast.error("First question # must be ≥ 1");
   391→        return;
   392→      }
   393→      const chap = chapter || "PDF questions";
   394→      meta = {
   395→        subject,
   396→        chapter: chap,
   397→        start_page: sp,
   398→        end_page: Math.max(sp, ep),
   399→        total_questions: tq,
   400→        key: keyLater ? [] : gridKey,
   401→        key_later: keyLater,
   402→        sections: [
   403→          {
   404→            subject,
   405→            chapter: chap,
   406→            first_q: fq,
   407→            last_q: fq + tq - 1,
   408→            start_page: sp,
   409→            end_page: Math.max(sp, ep),
   410→          },
   411→        ],
   412→        first_q: fq,
   413→        tolerance: tol,
   414→      };
   415→      subjectOrder = [subject];
   416→    }
   417→
   418→    if (!keyLater && gridKey.length === 0) {
   419→      toast.error("Answer key is empty — paste the key, upload a digital key PDF, fill the grid, or switch on “score later”");
   420→      return;
   421→    }
   422→    if (!keyLater && coverage.missing.length > 0) {
   423→      const ok = window.confirm(
   424→        `${coverage.missing.length} of ${paperTotal} questions have no key (${coverage.missing
   425→          .slice(0, 8)
   426→          .join(", ")}${coverage.missing.length > 8 ? "…" : ""}). They will be scored as unattempted. Start anyway?`
   427→      );
   428→      if (!ok) return;
   429→    }
   430→
   431→    const session: ActiveSession = {
   432→      key: "active",
   433→      test_id: uid(),
   434→      mode: "pdf",
   435→      test_type: "pdf",
   436→      name: name.trim() || "PDF test",
   437→      subject_order: subjectOrder,
   438→      question_ids: [],
   439→      pdf_meta: meta,
   440→      duration_min: Math.max(1, Number(duration) || 1),
   441→      started_at: Date.now(),
   442→      answers: {},
   443→      marked: [],
   444→      current: 0,
   445→      q_times: {},
   446→      q_entered_at: Date.now(),
   447→    };
   448→    await kvSet("pdf-blob", blob); // idempotent with the file-pick write
   449→    nav.startSession(session, blob);
   450→  }

   456→      <PageTitle
   457→        title="Page-Mode PDF Test"
   458→        subtitle="Scanned Arihant pages, NTA papers, coaching sheets — anything. Keep the NTA answer-key tab open while pasting keys."
   459→      />
   460→
   461→      <div className="grid lg:grid-cols-2 gap-6">
   462→        <div className="space-y-6">
   463→          <SectionCard title="1 · Upload PDF" subtitle="stays local — never uploaded anywhere">
   464→            <input
   465→              ref={fileRef}
   466→              type="file"
   467→              accept="application/pdf"
   468→              className="hidden"
   469→              onChange={(e) => {
   470→                const f = e.target.files?.[0];

   477→            >
   478→              <div className="text-sm font-medium text-stone-700">
   479→                {loading ? "Reading PDF…" : fileName ? `✓ ${fileName}` : "Click to choose a PDF file"}
   480→              </div>
   481→              {numPages > 0 ? (
   482→                <div className="text-xs text-stone-400 mt-1">{numPages} pages detected</div>

   485→                  rendered locally with pdf.js — no upload

   488→            </button>
   489→          </SectionCard>
   490→
   491→          <SectionCard
   492→            title="2 · Structure"
   493→            subtitle={mode === "single" ? "map one subject's questions to pages" : "full paper — one row per section"}
   494→          >
   495→            <div className="space-y-1.5 mb-4 max-w-xs">

   498→                value={mode}

   504→                  <SelectItem value="full">Full paper (3 sections)</SelectItem>
   505→                </SelectContent>
   506→              </Select>
   507→            </div>
   508→
   509→            {mode === "single" ? (
   510→              <>
   511→                <div className="grid grid-cols-2 gap-3">
   512→                  <div className="space-y-1.5">
   513→                    <Label className="text-xs">Subject</Label>
   514→                    <Select
   515→                      value={subject}
   516→                      onValueChange={(v) => {
   517→                        setSubject(v as Subject);
   518→                        setChapter("");
   519→                      }}
   520→                    >
   521→                      <SelectTrigger aria-label="Subject"><SelectValue /></SelectTrigger>
