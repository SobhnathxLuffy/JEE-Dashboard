    37→    37→/**
    38→    38→ * Answer-key text parser — tolerates the common dirty shapes:
    39→    39→ * "1. A", "1A", "1) b", "7. B or C", "7 B/C", "9. bonus", "1. 42", "1. -12.5".
    40→    40→ * Key numbers are paper numbering (whatever the sheet says).
    41→    41→ */
    42→    42→function parseAnswerKey(text: string): PdfKeyEntry[] {
    43→    43→  const out: PdfKeyEntry[] = [];
    44→    44→  const re =
    45→    45→    /(\d{1,3})\s*[.):\-]?\s*(bonus|(?:[A-Da-d]\s*(?:\/|,|[oO][rR]\b)\s*)+[A-Da-d]|[A-Da-d])(?![A-Za-z0-9/])|(\d{1,3})\s*[.):\-]\s*(-?\d+(?:\.\d+)?)/g;
    46→    46→  let m: RegExpExecArray | null;
    47→    47→  while ((m = re.exec(text)) !== null) {
    48→    48→    if (m[1] !== undefined && m[2] !== undefined) {
    49→    49→      const v = m[2].trim();
    50→    50→      if (/^bonus$/i.test(v)) {
    51→    51→        out.push({ no: Number(m[1]), answer: "", bonus: true });
    52→    52→      } else if (v.length > 1) {
    53→    53→        const letters = [
    54→    54→          ...new Set(
    55→    55→            v
    56→    56→              .split(/\s*(?:\/|,|[oO][rR]\b)\s*/)
    57→    57→              .filter(Boolean)
    58→    58→              .map((p) => p.toUpperCase())
    59→    59→          ),
    60→    60→        ].filter((p) => /^[A-D]$/.test(p));
    61→    61→        if (letters.length > 1) {
    62→    62→          out.push({ no: Number(m[1]), answer: letters[0], answers: letters });
    63→    63→        } else if (letters.length === 1) {
    64→    64→          out.push({ no: Number(m[1]), answer: letters[0] });
    65→    65→        }
    66→    66→      } else {
    67→    67→        out.push({ no: Number(m[1]), answer: v.toUpperCase() });
    68→    68→      }
    69→    69→    } else if (m[3] !== undefined && m[4] !== undefined) {
    70→    70→      out.push({ no: Number(m[3]), answer: m[4] });
    71→    71→    }
    72→    72→  }
    73→    73→  // dedupe by question number (last wins)
    74→    74→  const map = new Map<number, PdfKeyEntry>();
    75→    75→  for (const k of out) map.set(k.no, k);
    76→    76→  return [...map.values()].sort((a, b) => a.no - b.no);
    77→    77→}

    79→    79→type CellResult =
    80→    80→  | { kind: "empty" }
    81→    81→  | { kind: "entry"; entry: PdfKeyEntry }
    82→    82→  | { kind: "invalid" };

    84→    84→/** Interpret one key-grid cell (raw text) — option-number mapping applied here. */
    85→    85→function parseCell(raw: string, optionNums: boolean): CellResult {
    86→    86→  const t = raw.trim();
    87→    87→  if (t === "") return { kind: "empty" };
    88→    88→  if (/^bonus$/i.test(t)) {
    89→    89→    return { kind: "entry", entry: { no: 0, answer: "", bonus: true } };
    90→    90→  }
    91→    91→  const parts = t
    92→    92→    .split(/\s*(?:\/|,|[oO][rR]\b)\s*/)
    93→    93→    .filter(Boolean);
    94→    94→  if (parts.length > 1) {
    95→    95→    const letters = [
    96→    96→      ...new Set(parts.map((p) => p.toUpperCase())),
    97→    97→    ].filter((p) => /^[A-D]$/.test(p));
    98→    98→    if (letters.length > 1 && letters.length === parts.length) {
    99→    99→      return { kind: "entry", entry: { no: 0, answer: letters[0], answers: letters } };
   100→   100→    }
   101→   101→    return { kind: "invalid" };
   102→   102→  }
   103→   103→  if (/^[A-Da-d]$/.test(t)) {
   104→   104→    return { kind: "entry", entry: { no: 0, answer: t.toUpperCase() } };
   105→   105→  }
   106→   106→  if (optionNums && /^[1-4]$/.test(t)) {
   107→   107→    return { kind: "entry", entry: { no: 0, answer: LETTERS[Number(t) - 1] } };
   108→   108→  }
   109→   109→  if (NUM_RE.test(t)) {
   110→   110→    return { kind: "entry", entry: { no: 0, answer: t } };
   111→   111→  }
   112→   112→  return { kind: "invalid" };
   113→   113→}
