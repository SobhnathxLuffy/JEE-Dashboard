     1→"use client";
     2→
     3→// ─── Post-test analysis: score, accuracy, per-question review, error tags ───
     4→// Sprint D review side: paper-numbered questions + subject chips + next-untagged
     5→// jump (D3), retry-wrong as a new CBT session (D1), key-later self-mark with
     6→// live score recompute (D4), per-question notes (D7) and solution photos (D8).
     7→import { useMemo, useRef, useState } from "react";
     8→import { Button } from "@/components/ui/button";
     9→import { Input } from "@/components/ui/input";
    10→import { Badge } from "@/components/ui/badge";
    11→import {
    12→  Dialog,
    13→  DialogContent,
    14→  DialogDescription,
    15→  DialogFooter,
    16→  DialogHeader,
    17→  DialogTitle,
    18→} from "@/components/ui/dialog";
    19→import { toast } from "sonner";
    20→import { cn } from "@/lib/utils";
    21→import type { NavController } from "./App";
    22→import { EmptyNote, PageTitle, SectionCard, StatCard } from "./shared";
    23→import { useLive, put, get, getAll } from "@/lib/idb";
    24→import { marksFor } from "@/lib/scoring";
    25→import { fileToDataUrl } from "@/lib/image";
    26→import {
    27→  ERROR_TAGS,
    28→  SUBJECTS,
    29→  SUBJECT_SHORT,
    30→  fmtSecs,
    31→  uid,
    32→  type ActiveSession,
    33→  type ErrorTag,
    34→  type ResponseRecord,
    35→  type Subject,
    36→} from "@/lib/types";
    37→
    38→const TAG_CLS: Record<ErrorTag, string> = {
    39→  C: "bg-red-100 text-red-700 border-red-200",
    40→  F: "bg-amber-100 text-amber-800 border-amber-300",
    41→  A: "bg-orange-100 text-orange-700 border-orange-200",
    42→  R: "bg-sky-100 text-sky-700 border-sky-200",
    43→  T: "bg-violet-100 text-violet-700 border-violet-200",
    44→  G: "bg-stone-200 text-stone-700 border-stone-300",
    45→};
    46→
    47→type StatusFilter = "all" | "wrong" | "untagged";
    48→type SubjectFilter = "all" | Subject;
    49→
    50→export function ResultsView({ testId, nav }: { testId: string; nav: NavController }) {
    51→  const tests = useLive("tests");
    52→  const responses = useLive("responses");
    53→  const [filter, setFilter] = useState<StatusFilter>("all");
    54→  const [subjectFilter, setSubjectFilter] = useState<SubjectFilter>("all");
    55→
    56→  // D3 "next untagged" jump state
    57→  const [highlightId, setHighlightId] = useState<string | null>(null);
    58→  const jumpIdxRef = useRef(0);
    59→  const highlightTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    60→
    61→  // D7: first-save toast bookkeeping (per response, this visit)
    62→  const notedOnceRef = useRef<Set<string>>(new Set());
    63→
    64→  // D8: shared photo file input + full-size viewer dialog
    65→  const photoInputRef = useRef<HTMLInputElement>(null);
    66→  const photoTargetRef = useRef<string | null>(null);
    67→  const [viewPhoto, setViewPhoto] = useState<{ url: string; title: string; id: string } | null>(
    68→    null
    69→  );
    70→
    71→  const test = useMemo(() => tests.find((t) => t.id === testId), [tests, testId]);
    72→  const rows = useMemo(() => {
    73→    const own = responses.filter((r) => r.test_id === testId);
    74→    // review reads in paper/test order — q_no is the stored number, position is the fallback
    75→    return [...own].sort((a, b) => (a.q_no ?? 0) - (b.q_no ?? 0));
    76→  }, [responses, testId]);
    77→
    78→  // D4: key-later tests stay in self-mark mode for their whole life (score is
    79→  // recomputed to a number on the first toggle, so the flag is the discriminator)
    80→  const selfMark = test?.pdf_meta?.key_later === true;
    81→
    82→  // D3: displayed number = the paper/test number stored at submit (q_no),
    83→  // falling back to the row's stable position — never re-indexed by filters.
    84→  const qNoOf = useMemo(() => {
    85→    const m = new Map<string, number>();
    86→    rows.forEach((r, i) => m.set(r.id, r.q_no ?? i + 1));
    87→    return m;
    88→  }, [rows]);
    89→
    90→  const stats = useMemo(() => {
    91→    const attempted = rows.filter((r) => r.attempted);
    92→    const correct = rows.filter((r) => r.correct === true);
    93→    const wrong = rows.filter((r) => r.attempted && r.correct === false);
    94→    const untagged = wrong.filter((r) => !r.error_tag).length;
    95→    const time = rows.reduce((a, r) => a + (r.time_spent || 0), 0);
    96→    return {
    97→      attempted: attempted.length,
    98→      correct: correct.length,
    99→      wrong: wrong.length,
   100→      untagged,
   101→      time,
   102→      accuracy: attempted.length ? Math.round((correct.length / attempted.length) * 100) : 0,
   103→      attemptRate: rows.length ? Math.round((attempted.length / rows.length) * 100) : 0,
   104→      pending: attempted.filter((r) => r.correct === null).length,
   105→    };
   106→  }, [rows]);
   107→
   108→  const shown = useMemo(() => {
   109→    let out = rows;
   110→    if (subjectFilter !== "all") out = out.filter((r) => r.subject === subjectFilter);
   111→    if (filter === "wrong") out = out.filter((r) => r.attempted && r.correct === false);
   112→    if (filter === "untagged")
   113→      out = out.filter((r) => r.attempted && r.correct === false && !r.error_tag);
   114→    return out;
   115→  }, [rows, filter, subjectFilter]);
   116→
   117→  // D1: retry candidates — wrong + unattempted-but-keyed (key-later: attempted only)
   118→  const retryIds = useMemo(() => {
   119→    if (!test?.question_ids) return [];
   120→    const keyLater = test.pdf_meta?.key_later === true;
   121→    const candidates = rows.filter((r) => {
   122→      if (keyLater) return r.attempted;
   123→      if (r.attempted) return r.correct === false;
   124→      return r.correct_answer !== "?"; // skipped but keyed → retry it too
   125→    });
   126→    const want = new Set(candidates.map((r) => r.question_id));
   127→    return test.question_ids.filter((id) => want.has(id)); // test order preserved
   128→  }, [test, rows]);
   129→
   130→  function startRetry() {
   131→    if (!test || retryIds.length === 0) return;
   132→    const retryRows = rows.filter((r) => retryIds.includes(r.question_id));
   133→    const session: ActiveSession = {
   134→      key: "active",
   135→      test_id: uid(),
   136→      mode: "cbt",
   137→      test_type: "mixed",
   138→      name: `Retry — ${test.name}`,
   139→      subject_order: SUBJECTS.filter((s) => retryRows.some((r) => r.subject === s)),
   140→      question_ids: retryIds,
   141→      duration_min: Math.max(1, retryIds.length), // 1 min per question rule
   142→      started_at: Date.now(),
   143→      answers: {},
   144→      marked: [],
   145→      current: 0,
   146→      q_times: {},
   147→      q_entered_at: Date.now(),
   148→    };
   149→    nav.startSession(session);
   150→  }
   151→
   152→  // D3: cycle through wrong+untagged rows, scroll + highlight each in turn
   153→  const untaggedIds = useMemo(
   154→    () =>
   155→      rows
   156→        .filter((r) => r.attempted && r.correct === false && !r.error_tag)
   157→        .map((r) => r.id),
   158→    [rows]
   159→  );
   160→
   161→  function jumpToNextUntagged() {
   162→    if (untaggedIds.length === 0) return;
   163→    const idx = jumpIdxRef.current % untaggedIds.length;
   164→    jumpIdxRef.current += 1;
   165→    const id = untaggedIds[idx];
   166→    const el = document.getElementById(`resp-${id}`);
   167→    if (el) el.scrollIntoView({ behavior: "smooth", block: "center" });
   168→    setHighlightId(id);
   169→    if (highlightTimerRef.current) clearTimeout(highlightTimerRef.current);
   170→    highlightTimerRef.current = setTimeout(() => setHighlightId(null), 2500);
   171→  }
   172→
   173→  /** Fresh read + merge — the row prop may be stale when several edits race. */
   174→  async function patchResponse(id: string, patch: Partial<ResponseRecord>) {
   175→    const fresh = await get("responses", id);
   176→    if (!fresh) return null;
   177→    const next = { ...fresh, ...patch };
   178→    await put("responses", next);
   179→    return next;
   180→  }
   181→
   182→  async function tagResponse(r: ResponseRecord, tag: ErrorTag) {
   183→    const updated = await patchResponse(r.id, { error_tag: tag });
   184→    if (!updated) return;
   185→    if (tag === "F") {
   186→      const existing = await get("formula", `${updated.test_id}:${updated.question_id}`);
   187→      if (!existing) {
   188→        await put("formula", {
   189→          id: `${updated.test_id}:${updated.question_id}`,
   190→          test_id: updated.test_id,
   191→          question_id: updated.question_id,
   192→          subject: updated.subject,
   193→          chapter: updated.chapter,
   194→          snippet: updated.question_snippet,
   195→          created_at: Date.now(),
   196→        });
   197→        toast.success("Added to formula sheet (F-tag)");
   198→      }
   199→    } else {
   200→      // if retagged away from F, remove from formula sheet
   201→      const existing = await get("formula", `${updated.test_id}:${updated.question_id}`);
   202→      if (existing) {
   203→        const { del } = await import("@/lib/idb");
   204→        await del("formula", existing.id);
   205→      }
   206→    }
   207→  }
   208→
   209→  // D7: one-line note, saved on blur (Enter blurs). Toast on the first save of each note only.
   210→  async function saveNote(r: ResponseRecord, value: string) {
   211→    const v = value.trim();
   212→    if ((r.note ?? "") === v) return; // nothing changed
   213→    await patchResponse(r.id, { note: v === "" ? undefined : v });
   214→    if (!notedOnceRef.current.has(r.id)) {
   215→      notedOnceRef.current.add(r.id);
   216→      toast.success("Note saved");
   217→    }
   218→  }
   219→
   220→  // D8: photo attach/replace via the shared hidden file input
   221→  function pickPhoto(responseId: string) {
   222→    photoTargetRef.current = responseId;
   223→    photoInputRef.current?.click();
   224→  }
   225→
   226→  async function onPhotoFile(e: React.ChangeEvent<HTMLInputElement>) {
   227→    const file = e.target.files?.[0];
   228→    e.target.value = ""; // allow re-picking the same file later
   229→    const targetId = photoTargetRef.current;
   230→    photoTargetRef.current = null;
   231→    if (!file || !targetId) return;
   232→    try {
   233→      const dataUrl = await fileToDataUrl(file, 1000);
   234→      const updated = await patchResponse(targetId, { photo: dataUrl });
   235→      if (updated) toast.success("Solution photo attached");
   236→    } catch {
   237→      toast.error("Could not read that image — try a JPG/PNG");
   238→    }
   239→  }
   240→
   241→  async function removePhoto(id: string) {
   242→    await patchResponse(id, { photo: undefined });
   243→    setViewPhoto(null);
   244→    toast.success("Solution photo removed");
   245→  }
   246→
   247→  // D4: set the mark, then recompute the whole TestRecord from ALL responses
   248→  async function applyMark(r: ResponseRecord, correct: boolean | null) {
   249→    if (!test) return;
   250→    const updated = await patchResponse(r.id, { correct });
   251→    if (!updated) return;
   252→    const all = (await getAll("responses")).filter((x) => x.test_id === testId);
   253→    const subjectScores: Record<Subject, number> = {
   254→      Physics: 0,
   255→      Chemistry: 0,
   256→      Mathematics: 0,
   257→    };
   258→    for (const x of all) subjectScores[x.subject] += marksFor(x.attempted, x.correct);
   259→    const score = SUBJECTS.reduce((a, s) => a + subjectScores[s], 0);
   260→    const attemptedRows = all.filter((x) => x.attempted);
   261→    const allMarked =
   262→      attemptedRows.length > 0 && attemptedRows.every((x) => x.correct !== null);
   263→    const wasAllMarked = rows
   264→      .filter((x) => x.attempted)
   265→      .every((x) => x.correct !== null);
   266→    await put("tests", { ...test, score, subject_scores: subjectScores });
   267→    if (!wasAllMarked && allMarked) {
   268→      toast.success(`Score finalised: ${score}/${test.max_score}`);
   269→    }
   270→  }
   271→
   272→  if (!test) {
   273→    return (
   274→      <div className="space-y-4">
   275→        <PageTitle title="Test analysis" />
   276→        <EmptyNote>
   277→          Test not found.{" "}
   278→          <button className="underline" onClick={() => nav.go("dashboard")}>
   279→            Back to dashboard
   280→          </button>
   281→        </EmptyNote>
   282→      </div>
   283→    );
   284→  }
   285→
   286→  const markedCount = stats.attempted - stats.pending;
   287→
   288→  return (
   289→    <div className="space-y-6">
   290→      <PageTitle
   291→        title={test.name}
   292→        subtitle={`${test.date} · ${test.source} · ${test.duration_min} min · ${test.type} test`}
   293→        right={
   294→          <Button variant="outline" size="sm" onClick={() => nav.go("dashboard")}>
   295→            ← Dashboard
   296→          </Button>
   297→        }
   298→      />
   299→
   300→      <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   301→        <StatCard
   302→          label="Score"
   303→          value={test.score === null ? "pending" : `${test.score}/${test.max_score}`}
   304→          tone={(test.score ?? 0) >= test.max_score * 0.5 ? "good" : "warn"}
   305→          hint={
   306→            selfMark && stats.pending > 0
   307→              ? `${stats.pending} mark${stats.pending > 1 ? "s" : ""} pending`
   308→              : undefined
   309→          }
   310→        />
   311→        <StatCard label="Accuracy" value={`${stats.accuracy}%`} hint={`${stats.correct}/${stats.attempted}`} />
   312→        <StatCard label="Attempt rate" value={`${stats.attemptRate}%`} hint={`${stats.attempted}/${rows.length}`} />
   313→        <StatCard
   314→          label="Negatives"
   315→          value={`−${stats.wrong}`}
   316→          tone={stats.wrong > 5 ? "bad" : "default"}
   317→          hint={`score would be ${(test.score ?? 0) + stats.wrong} with guesses skipped`}
   318→        />
   319→        <StatCard label="Time used" value={fmtSecs(stats.time)} />
   320→      </div>
   321→
   322→      {/* D4: self-mark banner (key-later tests) */}
   323→      {selfMark ? (
   324→        stats.pending > 0 || test.score === null ? (
   325→          <div className="bg-amber-50 border border-amber-200 rounded-lg px-4 py-3 text-sm text-amber-800 flex items-center justify-between gap-3 flex-wrap">
   326→            <span>
   327→              <strong>Self-mark mode</strong> — mark each attempted question below. The score
   328→              recomputes as you go.
   329→            </span>
   330→            <Badge variant="outline" className="border-amber-400 text-amber-800 whitespace-nowrap">
   331→              pending marks: {markedCount}/{stats.attempted}
   332→            </Badge>
   333→          </div>
   334→        ) : (
   335→          <div className="bg-emerald-50 border border-emerald-200 rounded-lg px-4 py-3 text-sm text-emerald-800">
   336→            <strong>Self-mark complete</strong> — score {test.score}/{test.max_score}. Toggles
   337→            stay live if a mark needs correcting.
   338→          </div>
   339→        )
   340→      ) : null}
   341→
   342→      {/* D1: retry-wrong as a fresh CBT session */}
   343→      {test.question_ids ? (
   344→        <div className="bg-white border border-stone-200 rounded-lg px-4 py-3 flex items-center gap-3 flex-wrap">
   345→          <Button
   346→            onClick={startRetry}
   347→            disabled={retryIds.length === 0}
   348→            className="bg-emerald-700 hover:bg-emerald-800 min-h-[44px] px-5 font-semibold"
   349→          >
   350→            ↻ Retry wrong ({retryIds.length})
   351→          </Button>
   352→          <p className="text-xs text-stone-500 min-w-0 flex-1">
   353→            Wrong + skipped-but-keyed questions, test order, 1 min each. The fastest way to prove
   354→            the fix stuck.
   355→          </p>
   356→        </div>
   357→      ) : test.pdf_meta ? (
   358→        <div className="bg-white border border-stone-200 rounded-lg px-4 py-3 flex items-center gap-3 flex-wrap">
   359→          <Button disabled className="min-h-[44px] px-5">
   360→            ↻ Retry wrong
   361→          </Button>
   362→          <p className="text-xs text-stone-400">
   363→            PDF test — re-upload the PDF to retry this set.
   364→          </p>
   365→        </div>
   366→      ) : null}
   367→
   368→      {stats.untagged > 0 ? (
   369→        <div className="bg-amber-50 border border-amber-200 rounded-lg px-4 py-3 text-sm text-amber-800 flex items-center justify-between gap-3 flex-wrap">
   370→          <span>
   371→            <strong>{stats.untagged} wrong answers</strong> still untagged. Tag them now — the
   372→            dashboard&apos;s error-tag chart and formula sheet only work if every mistake is
   373→            classified.
   374→          </span>
   375→          <div className="flex gap-2">
   376→            <Button
   377→              size="sm"
   378→              variant="outline"
   379→              className="border-amber-400 text-amber-800 hover:bg-amber-100"
   380→              onClick={jumpToNextUntagged}
   381→            >
   382→              ↦ Next untagged ({untaggedIds.length})
   383→            </Button>
   384→            <Button
   385→              size="sm"
   386→              variant="outline"
   387→              className="border-amber-400 text-amber-800 hover:bg-amber-100"
   388→              onClick={() => setFilter("untagged")}
   389→            >
   390→              Show untagged
   391→            </Button>
   392→          </div>
   393→        </div>
   394→      ) : null}
   395→
   396→      <SectionCard
   397→        title="Per-question review"
   398→        subtitle="your answer vs correct, time spent, and the tag that explains the mistake"
   399→        action={
   400→          <div className="flex flex-col items-end gap-1.5">
   401→            <div className="flex gap-1.5">
   402→              {(["all", "wrong", "untagged"] as const).map((f) => (
   403→                <button
   404→                  key={f}
   405→                  onClick={() => setFilter(f)}
   406→                  className={cn(
   407→                    "px-2.5 py-1 rounded-full text-xs transition-colors",
   408→                    filter === f
   409→                      ? "bg-stone-900 text-white"
   410→                      : "bg-stone-100 text-stone-500 hover:bg-stone-200"
   411→                  )}
   412→                >
   413→                  {f}
   414→                </button>
   415→              ))}
   416→            </div>
   417→            <div className="flex gap-1.5" role="group" aria-label="Filter by subject">
   418→              {(["all", "Physics", "Chemistry", "Mathematics"] as const).map((s) => (
   419→                <button
   420→                  key={s}
   421→                  onClick={() => setSubjectFilter(s)}
   422→                  aria-pressed={subjectFilter === s}
   423→                  className={cn(
   424→                    "px-2.5 py-1 rounded-full text-xs transition-colors min-h-[28px]",
   425→                    subjectFilter === s
   426→                      ? "bg-emerald-700 text-white font-medium"
   427→                      : "bg-stone-100 text-stone-500 hover:bg-stone-200"
   428→                  )}
   429→                >
   430→                  {s === "all" ? "All" : SUBJECT_SHORT[s]}
   431→                </button>
   432→              ))}
   433→            </div>
   434→          </div>
   435→        }
   436→      >
   437→        {/* D8 shared hidden input — one file picker serves every row */}
   438→        <input
   439→          ref={photoInputRef}
   440→          type="file"
   441→          accept="image/*"
   442→          className="hidden"
   443→          onChange={(e) => void onPhotoFile(e)}
   444→        />
   445→
   446→        {shown.length === 0 ? (
   447→          <EmptyNote>Nothing to show for this filter.</EmptyNote>
   448→        ) : (
   449→          <ul className="space-y-2 max-h-[600px] overflow-y-auto pr-1">
   450→            {shown.map((r) => {
   451→              const qNo = qNoOf.get(r.id) ?? 1;
   452→              return (
   453→                <li
   454→                  key={r.id}
   455→                  id={`resp-${r.id}`}
   456→                  className={cn(
   457→                    "border rounded-lg p-3 transition-shadow",
   458→                    highlightId === r.id
   459→                      ? "border-amber-400 bg-amber-50 ring-2 ring-amber-300"
   460→                      : !r.attempted
   461→                        ? "border-stone-100 bg-stone-50"
   462→                        : r.correct === true
   463→                          ? "border-emerald-100 bg-emerald-50/50"
   464→                          : r.correct === false
   465→                            ? "border-red-100 bg-red-50/40"
   466→                            : "border-amber-100 bg-amber-50/40" // pending self-mark
   467→                  )}
   468→                >
   469→                  <div className="flex items-start justify-between gap-3 flex-wrap">
   470→                    <div className="min-w-0 flex-1">
   471→                      <div className="flex items-center gap-2 mb-1 flex-wrap">
   472→                        <span className="text-xs font-bold text-stone-500">
   473→                          Q{qNo}
   474→                        </span>
   475→                        <Badge variant="outline" className="text-[10px] border-stone-300 text-stone-500">
   476→                          {r.type === "numerical" ? "NUM" : "MCQ"}
   477→                        </Badge>
   478→                        <span className="text-[11px] text-stone-400">{r.chapter}</span>
   479→                        <span className="text-[11px] text-stone-400">{fmtSecs(r.time_spent)}</span>
   480→                        {!r.attempted ? (
   481→                          <Badge variant="outline" className="text-[10px] border-stone-300 text-stone-400">
   482→                            unattempted
   483→                          </Badge>
   484→                        ) : r.correct === true ? (
   485→                          <Badge className="text-[10px] bg-emerald-600 hover:bg-emerald-600 text-white border-0">
   486→                            +4
   487→                          </Badge>
   488→                        ) : r.correct === false ? (
   489→                          <Badge className="text-[10px] bg-red-600 hover:bg-red-600 text-white border-0">
   490→                            −1
   491→                          </Badge>
   492→                        ) : (
   493→                          <Badge className="text-[10px] bg-amber-500 hover:bg-amber-500 text-white border-0">
   494→                            pending
   495→                          </Badge>
   496→                        )}
   497→                      </div>
   498→                      <p className="text-sm text-stone-800">{r.question_snippet}</p>
   499→                      <div className="flex flex-wrap gap-x-4 gap-y-1 text-xs mt-1.5">
   500→                        <span className={r.correct === true ? "text-emerald-700" : "text-red-600"}>
   501→                          You:{" "}
   502→                          <strong>
   503→                            {r.attempted
   504→                              ? r.type === "MCQ"
   505→                                ? ["A", "B", "C", "D"][Number(r.selected)] ?? String(r.selected)
   506→                                : String(r.selected)
   507→                              : "—"}
   508→                          </strong>
   509→                        </span>
   510→                        <span className="text-emerald-700">
   511→                          Correct:{" "}
   512→                          <strong>
   513→                            {r.type === "MCQ"
   514→                              ? ["A", "B", "C", "D"][Number(r.correct_answer)] ?? String(r.correct_answer)
   515→                              : String(r.correct_answer)}
   516→                          </strong>
   517→                        </span>
   518→                      </div>
   519→
   520→                      {/* D7: one-line note (optional, blur-save) */}
   521→                      <Input
   522→                        defaultValue={r.note ?? ""}
   523→                        placeholder="Add a one-line note — the actual insight (optional)"
   524→                        className="mt-2 h-8 text-xs bg-white"
   525→                        maxLength={280}
   526→                        aria-label={`Note for Q${qNo}`}
   527→                        onBlur={(e) => void saveNote(r, e.target.value)}
   528→                        onKeyDown={(e) => {
   529→                          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
   530→                        }}
   531→                      />
   532→
   533→                      {/* D8: solution photo — thumbnail opens full size */}
   534→                      {r.photo ? (
   535→                        <button
   536→                          type="button"
   537→                          onClick={() => setViewPhoto({ url: r.photo!, title: `Q${qNo} · ${r.chapter}`, id: r.id })}
   538→                          className="mt-2 block rounded-md overflow-hidden border border-stone-300 hover:border-emerald-500 transition-colors"
   539→                          aria-label={`View solution photo for Q${qNo}`}
   540→                        >
   541→                          <img
   542→                            src={r.photo}
   543→                            alt={`Solution photo for Q${qNo}`}
   544→                            className="h-14 w-20 object-cover"
   545→                          />
   546→                        </button>
   547→                      ) : (
   548→                        <Button
   549→                          type="button"
   550→                          variant="outline"
   551→                          size="sm"
   552→                          className="mt-2 h-8 text-xs"
   553→                          onClick={() => pickPhoto(r.id)}
   554→                        >
   555→                          📎 Attach solution photo
   556→                        </Button>
   557→                      )}
   558→                    </div>
   559→
   560→                    {/* D4: self-mark toggles (key-later tests only) */}
   561→                    {selfMark ? (
   562→                      <div className="flex flex-col gap-1 items-end shrink-0">
   563→                        <span className="text-[10px] uppercase text-stone-400 font-medium">mark it</span>
   564→                        <div className="flex gap-1.5">
   565→                          <button
   566→                            type="button"
   567→                            onClick={() => void applyMark(r, r.correct === true ? null : true)}
   568→                            className={cn(
   569→                              "px-2.5 min-h-[36px] rounded-md border text-xs font-semibold transition-colors",
   570→                              r.correct === true
   571→                                ? "bg-emerald-700 text-white border-emerald-700"
   572→                                : "bg-white border-stone-300 text-stone-600 hover:border-emerald-500"
   573→                            )}
   574→                          >
   575→                            ✓ Correct
   576→                          </button>
   577→                          <button
   578→                            type="button"
   579→                            onClick={() => void applyMark(r, r.correct === false ? null : false)}
   580→                            className={cn(
   581→                              "px-2.5 min-h-[36px] rounded-md border text-xs font-semibold transition-colors",
   582→                              r.correct === false
   583→                                ? "bg-red-600 text-white border-red-600"
   584→                                : "bg-white border-stone-300 text-stone-600 hover:border-red-400"
   585→                            )}
   586→                          >
   587→                            ✗ Wrong
   588→                          </button>
   589→                        </div>
   590→                      </div>
   591→                    ) : null}
   592→
   593→                    {/* tag UI: wrong-marked rows only (self-marked wrongs flow in here too) */}
   594→                    {r.attempted && r.correct === false ? (
   595→                      <div className="flex flex-col gap-1 items-end shrink-0">
   596→                        <span className="text-[10px] uppercase text-stone-400 font-medium">tag it</span>
   597→                        <div className="flex gap-1">
   598→                          {ERROR_TAGS.map((t) => (
   599→                            <button
   600→                              key={t.code}
   601→                              title={t.hint}
   602→                              onClick={() => void tagResponse(r, t.code)}
   603→                              className={cn(
   604→                                "w-7 h-7 rounded-md border text-xs font-bold transition-colors",
   605→                                r.error_tag === t.code
   606→                                  ? TAG_CLS[t.code] + " ring-2 ring-offset-1 ring-stone-300"
   607→                                  : "bg-white border-stone-200 text-stone-400 hover:text-stone-700 hover:border-stone-400"
   608→                              )}
   609→                            >
   610→                              {t.code}
   611→                            </button>
   612→                          ))}
   613→                        </div>
   614→                        {r.error_tag ? (
   615→                          <span className="text-[10px] text-stone-400">
   616→                            {ERROR_TAGS.find((t) => t.code === r.error_tag)?.label}
   617→                            {r.error_tag === "F" ? " → formula sheet ✓" : ""}
   618→                          </span>
   619→                        ) : null}
   620→                      </div>
   621→                    ) : null}
   622→                  </div>
   623→                </li>
   624→              );
   625→            })}
   626→          </ul>
   627→        )}
   628→      </SectionCard>
   629→
   630→      {/* D8: full-size solution photo viewer */}
   631→      <Dialog open={viewPhoto !== null} onOpenChange={(open) => !open && setViewPhoto(null)}>
   632→        <DialogContent className="max-w-2xl">
   633→          <DialogHeader>
   634→            <DialogTitle>Solution photo — {viewPhoto?.title}</DialogTitle>
   635→            <DialogDescription>
   636→              Attached to this question&apos;s review. Replace or remove it below.
   637→            </DialogDescription>
   638→          </DialogHeader>
   639→          {viewPhoto ? (
   640→            <img
   641→              src={viewPhoto.url}
   642→              alt={`Solution photo, ${viewPhoto.title}`}
   643→              className="w-full h-auto max-h-[65vh] object-contain rounded-md border border-stone-200 bg-white"
   644→            />
   645→          ) : null}
   646→          <DialogFooter className="gap-2 sm:gap-0">
   647→            <Button
   648→              variant="outline"
   649→              size="sm"
   650→              onClick={() => {
   651→                if (viewPhoto) pickPhoto(viewPhoto.id);
   652→                setViewPhoto(null);
   653→              }}
   654→            >
   655→              Replace
   656→            </Button>
   657→            <Button
   658→              variant="ghost"
   659→              size="sm"
   660→              className="text-red-500 hover:text-red-600 hover:bg-red-50"
   661→              onClick={() => viewPhoto && void removePhoto(viewPhoto.id)}
   662→            >
   663→              Remove
   664→            </Button>
   665→          </DialogFooter>
   666→        </DialogContent>
   667→      </Dialog>
   668→    </div>
   669→  );
   670→}
