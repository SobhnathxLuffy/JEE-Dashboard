    59→/** E2: consecutive days where all four blocks are true — ends today if today is
    60→ *  complete, otherwise yesterday (today still in progress never breaks it). */
    61→function computeStreak(logs: DailyLog[], today: string): number {
    62→  const byDate = new Map(logs.map((l) => [l.date, l]));
    63→  const allDone = (l: DailyLog | undefined) => !!l && Object.values(l.blocks).every(Boolean);
    64→  let cursor = allDone(byDate.get(today)) ? today : addDays(today, -1);
    65→  let streak = 0;
    66→  while (allDone(byDate.get(cursor))) {
    67→    streak += 1;
    68→    cursor = addDays(cursor, -1);
    69→  }
    70→  return streak;
    71→}

    74→function TagPctTooltip({

    76→  payload,
    77→}: {
    78→  active?: boolean;
    79→  payload?: { payload?: { tag?: string; count?: number; pct?: number } }[];
    80→}) {
    81→  if (!active || !payload || payload.length === 0 || !payload[0]?.payload) return null;
    82→  const d = payload[0].payload;
    83→  return (
    84→    <div className="rounded-lg border border-stone-200 bg-white px-2.5 py-1.5 text-xs shadow-sm">
    85→      <div className="font-medium text-stone-800">{d.tag}</div>

    87→        {d.count} wrong · {d.pct}% of tagged
    88→      </div>
    89→    </div>
    90→  );
    91→}

   644→// ─── Today card ──────────────────────────────────────────────────────────────
   645→const BLOCKS: { key: keyof DailyLog["blocks"]; label: string }[] = [
   646→  { key: "math", label: "Math — 2h" },
   647→  { key: "physics", label: "Physics — 1h 45m" },
   648→  { key: "chemistry", label: "Chemistry — 1h 45m" },
   649→  { key: "recall", label: "Recall + errors — 30m" },
   650→];
   651→
   652→function TodayCard({
   653→  log,
   654→  date,
   655→  streak,
   656→}: {
   657→  log: DailyLog | undefined;
   658→  date: string;
   659→  streak: number;
   660→}) {
   661→  const [chapters, setChapters] = useState(log?.chapters ?? "");
   662→  const [editing, setEditing] = useState(false);
   663→
   664→  // E5: read the FRESHEST row before writing — a block toggle must never
   665→  // clobber a concurrent write or the unsaved "tonight's chapters" text.
   666→  // The daily_log object store is keyed "id" (like every non-kv store), so
   667→  // rows are persisted as { id: date, ...log } — a bare {date,…} row throws
   668→  // DataError on put (pre-existing bug: the old Today card never persisted).
   669→  async function freshRow(): Promise<DailyLog & { id: string }> {
   670→    const row = await get("daily_log", date);
   671→    if (row) return { ...row, id: date };
   672→    return {
   673→      id: date,
   674→      date,
   675→      blocks: { math: false, physics: false, chemistry: false, recall: false },
   676→      chapters: "",
   677→    };
   678→  }
   679→
   680→  async function toggle(key: keyof DailyLog["blocks"]) {
   681→    const fresh = await freshRow();
   682→    const next: DailyLog & { id: string } = {
   683→      ...fresh,
   684→      id: date,
   685→      blocks: { ...fresh.blocks, [key]: !fresh.blocks[key] },
   686→      // carry unsaved chapter text along instead of wiping it
   687→      chapters: chapters.trim() ? chapters : fresh.chapters,
   688→    };
   689→    await put("daily_log", next);
   690→  }
   691→
   692→  async function saveChapters() {
   693→    const fresh = await freshRow();
   694→    const next: DailyLog & { id: string } = { ...fresh, id: date, chapters, date };
   695→    await put("daily_log", next);
   696→    setEditing(false);
   697→  }
   698→
   699→  const done = log ? BLOCKS.filter((b) => log.blocks[b.key]).length : 0;
   700→
   701→  return (
   702→    <SectionCard
   703→      title="Today"
   704→      subtitle={date}
   705→      action={
   706→        <div className="flex items-center gap-1.5">
   707→          {/* E2: streak chip — plain text, subtle */}
   708→          {streak >= 1 ? (
   709→            <Badge
   710→              variant="outline"
   711→              className={
   712→                streak >= 3
   713→                  ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   714→                  : "border-stone-300 text-stone-600"
   715→              }
   716→            >
   717→              {streak}-day streak
   718→            </Badge>
   719→          ) : null}
   720→          <Badge
   721→            variant="outline"
   722→            className={
   723→              done === 4
   724→                ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   725→                : "border-stone-300 text-stone-500"
   726→            }
   727→          >
   728→            {done}/4 blocks
   729→          </Badge>
   730→        </div>
   731→      }
   732→    >
   733→      <ul className="space-y-2">
   734→        {BLOCKS.map((b) => (
   735→          <li key={b.key}>
   736→            <label className="flex items-center gap-3 rounded-lg border border-stone-200 px-3 py-2.5 cursor-pointer hover:bg-stone-50 transition-colors">
   737→              <Checkbox
   738→                checked={log?.blocks[b.key] ?? false}
   739→                onCheckedChange={() => void toggle(b.key)}
   740→                aria-label={b.label}
   741→              />
   742→              <span
   743→                className={
   744→                  log?.blocks[b.key]
   745→                    ? "text-sm text-stone-400 line-through"
   746→                    : "text-sm text-stone-800"
   747→                }
   748→              >
   749→                {b.label}
   750→              </span>
   751→            </label>
   752→          </li>
   753→        ))}
   754→      </ul>
   755→      <div className="mt-3">
   756→        <div className="text-[11px] uppercase tracking-wide text-stone-400 font-medium mb-1">
   757→          Tonight&apos;s chapters
   758→        </div>
   759→        {editing ? (
   760→          <div className="flex gap-2">
   761→            <Input
   762→              value={chapters}
   763→              onChange={(e) => setChapters(e.target.value)}
   764→              placeholder="e.g. Electrostatics PYQs, GOC notes, 1-3-7 rev of Kinematics"
   765→              onKeyDown={(e) => {
   766→                if (e.key === "Enter") void saveChapters();
   767→              }}
   768→              onBlur={() => void saveChapters()}
   769→              autoFocus
   770→            />
   771→            <Button size="sm" onClick={() => void saveChapters()} className="bg-emerald-700 hover:bg-emerald-800">
   772→              Save
   773→            </Button>
   774→          </div>
   775→        ) : (
   776→          <button
   777→            className="w-full text-left text-sm rounded-lg border border-dashed border-stone-300 px-3 py-2.5 text-stone-600 hover:bg-stone-50 transition-colors"
   778→            onClick={() => {
   779→              setChapters(log?.chapters ?? "");
   780→              setEditing(true);
   781→            }}
   782→          >
   783→            {log?.chapters ? log.chapters : "Click to write what you will physically do tonight…"}
   784→          </button>
   785→        )}
   786→      </div>
   787→      <p className="text-[11px] text-stone-400 mt-3">
   788→        A plan fails if Tuesday night arrives and you don&apos;t know what to physically do. This
   789→        card answers that in one glance.
   790→      </p>
   791→    </SectionCard>
   792→  );
   793→}
