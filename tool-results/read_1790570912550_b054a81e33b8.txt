    59→    59→/** E2: consecutive days where all four blocks are true — ends today if today is
    60→    60→ *  complete, otherwise yesterday (today still in progress never breaks it). */
    61→    61→function computeStreak(logs: DailyLog[], today: string): number {
    62→    62→  const byDate = new Map(logs.map((l) => [l.date, l]));
    63→    63→  const allDone = (l: DailyLog | undefined) => !!l && Object.values(l.blocks).every(Boolean);
    64→    64→  let cursor = allDone(byDate.get(today)) ? today : addDays(today, -1);
    65→    65→  let streak = 0;
    66→    66→  while (allDone(byDate.get(cursor))) {
    67→    67→    streak += 1;
    68→    68→    cursor = addDays(cursor, -1);
    69→    69→  }
    70→    70→  return streak;
    71→    71→}

   651→   651→
   652→   652→function TodayCard({
   653→   653→  log,
   654→   654→  date,
   655→   655→  streak,
   656→   656→}: {
   657→   657→  log: DailyLog | undefined;
   658→   658→  date: string;
   659→   659→  streak: number;
   660→   660→}) {
   661→   661→  const [chapters, setChapters] = useState(log?.chapters ?? "");
   662→   662→  const [editing, setEditing] = useState(false);
   663→   663→
   664→   664→  // E5: read the FRESHEST row before writing — a block toggle must never
   665→   665→  // clobber a concurrent write or the unsaved "tonight's chapters" text.
   666→   666→  // The daily_log object store is keyed "id" (like every non-kv store), so
   667→   667→  // rows are persisted as { id: date, ...log } — a bare {date,…} row throws
   668→   668→  // DataError on put (pre-existing bug: the old Today card never persisted).
   669→   669→  async function freshRow(): Promise<DailyLog & { id: string }> {
   670→   670→    const row = await get("daily_log", date);
   671→   671→    if (row) return { ...row, id: date };
   672→   672→    return {
   673→   673→      id: date,
   674→   674→      date,
   675→   675→      blocks: { math: false, physics: false, chemistry: false, recall: false },
   676→   676→      chapters: "",
   677→   677→    };
   678→   678→  }
   679→   679→
   680→   680→  async function toggle(key: keyof DailyLog["blocks"]) {
   681→   681→    const fresh = await freshRow();
   682→   682→    const next: DailyLog & { id: string } = {
   683→   683→      ...fresh,
   684→   684→      id: date,
   685→   685→      blocks: { ...fresh.blocks, [key]: !fresh.blocks[key] },
   686→   686→      // carry unsaved chapter text along instead of wiping it
   687→   687→      chapters: chapters.trim() ? chapters : fresh.chapters,
   688→   688→    };
   689→   689→    await put("daily_log", next);
   690→   690→  }
   691→   691→
   692→   692→  async function saveChapters() {
   693→   693→    const fresh = await freshRow();
   694→   694→    const next: DailyLog & { id: string } = { ...fresh, id: date, chapters, date };
   695→   695→    await put("daily_log", next);
   696→   696→    setEditing(false);
   697→   697→  }
   698→   698→
   699→   699→  const done = log ? BLOCKS.filter((b) => log.blocks[b.key]).length : 0;
   700→   700→
   701→   701→  return (
   702→   702→    <SectionCard
   703→   703→      title="Today"
   704→   704→      subtitle={date}
   705→   705→      action={
   706→   706→        <div className="flex items-center gap-1.5">
   707→   707→          {/* E2: streak chip — plain text, subtle */}
   708→   708→          {streak >= 1 ? (
   709→   709→            <Badge
   710→   710→              variant="outline"
   711→   711→              className={
   712→   712→                streak >= 3
   713→   713→                  ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   714→   714→                  : "border-stone-300 text-stone-600"
   715→   715→              }
   716→   716→            >
   717→   717→              {streak}-day streak
   718→   718→            </Badge>
   719→   719→          ) : null}
   720→   720→          <Badge
   721→   721→            variant="outline"
   722→   722→            className={
   723→   723→              done === 4
   724→   724→                ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   725→   725→                : "border-stone-300 text-stone-500"
   726→   726→            }
   727→   727→          >
   728→   728→            {done}/4 blocks
   729→   729→          </Badge>
   730→   730→        </div>
   731→   731→      }
   732→   732→    >
   733→   733→      <ul className="space-y-2">
   734→   734→        {BLOCKS.map((b) => (
   735→   735→          <li key={b.key}>
   736→   736→            <label className="flex items-center gap-3 rounded-lg border border-stone-200 px-3 py-2.5 cursor-pointer hover:bg-stone-50 transition-colors">
   737→   737→              <Checkbox
   738→   738→                checked={log?.blocks[b.key] ?? false}
   739→   739→                onCheckedChange={() => void toggle(b.key)}
   740→   740→                aria-label={b.label}
   741→   741→              />
   742→   742→              <span
   743→   743→                className={
   744→   744→                  log?.blocks[b.key]
   745→   745→                    ? "text-sm text-stone-400 line-through"
   746→   746→                    : "text-sm text-stone-800"
   747→   747→                }
   748→   748→              >
   749→   749→                {b.label}
   750→   750→              </span>
   751→   751→            </label>
   752→   752→          </li>
   753→   753→        ))}
   754→   754→      </ul>
   755→   755→      <div className="mt-3">
   756→   756→        <div className="text-[11px] uppercase tracking-wide text-stone-400 font-medium mb-1">
   757→   757→          Tonight&apos;s chapters
   758→   758→        </div>
   759→   759→        {editing ? (
   760→   760→          <div className="flex gap-2">
   761→   761→            <Input
   762→   762→              value={chapters}
   763→   763→              onChange={(e) => setChapters(e.target.value)}

   765→   765→              onKeyDown={(e) => {
   766→   766→                if (e.key === "Enter") void saveChapters();
   767→   767→              }}
   768→   768→              onBlur={() => void saveChapters()}
   769→   769→              autoFocus
   770→   770→            />
   771→   771→            <Button size="sm" onClick={() => void saveChapters()} className="bg-emerald-700 hover:bg-emerald-800">
   772→   772→              Save
   773→   773→            </Button>
   774→   774→          </div>
   775→   775→        ) : (
   776→   776→          <button
   777→   777→            className="w-full text-left text-sm rounded-lg border border-dashed border-stone-300 px-3 py-2.5 text-stone-600 hover:bg-stone-50 transition-colors"
   778→   778→            onClick={() => {
   779→   779→              setChapters(log?.chapters ?? "");
   780→   780→              setEditing(true);
   781→   781→            }}
   782→   782→          >
   783→   783→            {log?.chapters ? log.chapters : "Click to write what you will physically do tonight…"}
   784→   784→          </button>

   791→   791→    </SectionCard>
   792→   792→  );
   793→   793→}
