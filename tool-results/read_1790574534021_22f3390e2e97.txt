    46→/** E2: consecutive days where all four blocks are true — ends today if today is
    47→ *  complete, otherwise yesterday (today still in progress never breaks it). */
    48→function computeStreak(logs: DailyLog[], today: string): number {
    49→  const byDate = new Map(logs.map((l) => [l.date, l]));
    50→  const allDone = (l: DailyLog | undefined) => !!l && Object.values(l.blocks).every(Boolean);
    51→  let cursor = allDone(byDate.get(today)) ? today : addDays(today, -1);
    52→  let streak = 0;
    53→  while (allDone(byDate.get(cursor))) {
    54→    streak += 1;
    55→    cursor = addDays(cursor, -1);
    56→  }
    57→  return streak;
    58→}

    60→/** E3b: tooltip for the error-tags bar — raw count + share of tagged wrong. */
    61→function TagPctTooltip({
    62→  active,
    63→  payload,
    64→}: {
    65→  active?: boolean;
    66→  payload?: { payload?: { tag?: string; count?: number; pct?: number } }[];
    67→}) {
    68→  if (!active || !payload || payload.length === 0 || !payload[0]?.payload) return null;
    69→  const d = payload[0].payload;
    70→  return (
    71→    <div className="rounded-lg border border-stone-200 bg-white px-2.5 py-1.5 text-xs shadow-sm">
    72→      <div className="font-medium text-stone-800">{d.tag}</div>
    73→      <div className="text-stone-500 tabular-nums">
    74→        {d.count} wrong · {d.pct}% of tagged
    75→      </div>
    76→    </div>
    77→  );
    78→}

   631→// ─── Today card ──────────────────────────────────────────────────────────────
   632→const BLOCKS: { key: keyof DailyLog["blocks"]; label: string }[] = [
   633→  { key: "math", label: "Math — 2h" },
   634→  { key: "physics", label: "Physics — 1h 45m" },
   635→  { key: "chemistry", label: "Chemistry — 1h 45m" },
   636→  { key: "recall", label: "Recall + errors — 30m" },
   637→];
   638→
   639→function TodayCard({
   640→  log,
   641→  date,
   642→  streak,
   643→}: {
   644→  log: DailyLog | undefined;
   645→  date: string;
   646→  streak: number;
   647→}) {
   648→  const [chapters, setChapters] = useState(log?.chapters ?? "");
   649→  const [editing, setEditing] = useState(false);
   650→
   651→  // E5: read the FRESHEST row before writing — a block toggle must never
   652→  // clobber a concurrent write or the unsaved "tonight's chapters" text.
   653→  // The daily_log object store is keyed "id" (like every non-kv store), so
   654→  // rows are persisted as { id: date, ...log } — a bare {date,…} row throws
   655→  // DataError on put (pre-existing bug: the old Today card never persisted).
   656→  async function freshRow(): Promise<DailyLog & { id: string }> {
   657→    const row = await get("daily_log", date);
   658→    if (row) return { ...row, id: date };
   659→    return {
   660→      id: date,
   661→      date,
   662→      blocks: { math: false, physics: false, chemistry: false, recall: false },
   663→      chapters: "",
   664→    };
   665→  }
   666→
   667→  async function toggle(key: keyof DailyLog["blocks"]) {
   668→    const fresh = await freshRow();
   669→    const next: DailyLog & { id: string } = {
   670→      ...fresh,
   671→      id: date,
   672→      blocks: { ...fresh.blocks, [key]: !fresh.blocks[key] },
   673→      // carry unsaved chapter text along instead of wiping it
   674→      chapters: chapters.trim() ? chapters : fresh.chapters,
   675→    };
   676→    await put("daily_log", next);
   677→  }
   678→
   679→  async function saveChapters() {
   680→    const fresh = await freshRow();
   681→    const next: DailyLog & { id: string } = { ...fresh, id: date, chapters, date };
   682→    await put("daily_log", next);
   683→    setEditing(false);
   684→  }
   685→
   686→  const done = log ? BLOCKS.filter((b) => log.blocks[b.key]).length : 0;
   687→
   688→  return (
   689→    <SectionCard
   690→      title="Today"
   691→      subtitle={date}
   692→      action={
   693→        <div className="flex items-center gap-1.5">
   694→          {/* E2: streak chip — plain text, subtle */}
   695→          {streak >= 1 ? (
   696→            <Badge
   697→              variant="outline"
   698→              className={
   699→                streak >= 3
   700→                  ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   701→                  : "border-stone-300 text-stone-600"
   702→              }
   703→            >
   704→              {streak}-day streak
   705→            </Badge>
   706→          ) : null}
   707→          <Badge
   708→            variant="outline"
   709→            className={
   710→              done === 4
   711→                ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   712→                : "border-stone-300 text-stone-500"
   713→            }
   714→          >
   715→            {done}/4 blocks
   716→          </Badge>
   717→        </div>
   718→      }
   719→    >
   720→      <ul className="space-y-2">
   721→        {BLOCKS.map((b) => (
   722→          <li key={b.key}>
   723→            <label className="flex items-center gap-3 rounded-lg border border-stone-200 px-3 py-2.5 cursor-pointer hover:bg-stone-50 transition-colors">
   724→              <Checkbox
   725→                checked={log?.blocks[b.key] ?? false}
   726→                onCheckedChange={() => void toggle(b.key)}
   727→                aria-label={b.label}
   728→              />
   729→              <span
   730→                className={
   731→                  log?.blocks[b.key]
   732→                    ? "text-sm text-stone-400 line-through"
   733→                    : "text-sm text-stone-800"
   734→                }
   735→              >
   736→                {b.label}
   737→              </span>
   738→            </label>
   739→          </li>
   740→        ))}
   741→      </ul>
   742→      <div className="mt-3">
   743→        <div className="text-[11px] uppercase tracking-wide text-stone-400 font-medium mb-1">
   744→          Tonight&apos;s chapters
   745→        </div>
   746→        {editing ? (
   747→          <div className="flex gap-2">
   748→            <Input
   749→              value={chapters}
   750→              onChange={(e) => setChapters(e.target.value)}
   751→              placeholder="e.g. Electrostatics PYQs, GOC notes, 1-3-7 rev of Kinematics"
   752→              onKeyDown={(e) => {
   753→                if (e.key === "Enter") void saveChapters();
   754→              }}
   755→              onBlur={() => void saveChapters()}
   756→              autoFocus
   757→            />
   758→            <Button size="sm" onClick={() => void saveChapters()} className="bg-emerald-700 hover:bg-emerald-800">
   759→              Save
   760→            </Button>
   761→          </div>
   762→        ) : (
   763→          <button
   764→            className="w-full text-left text-sm rounded-lg border border-dashed border-stone-300 px-3 py-2.5 text-stone-600 hover:bg-stone-50 transition-colors"
   765→            onClick={() => {
   766→              setChapters(log?.chapters ?? "");
   767→              setEditing(true);
   768→            }}
   769→          >
   770→            {log?.chapters ? log.chapters : "Click to write what you will physically do tonight…"}
   771→          </button>
   772→        )}
   773→      </div>
   774→      <p className="text-[11px] text-stone-400 mt-3">
   775→        A plan fails if Tuesday night arrives and you don&apos;t know what to physically do. This
   776→        card answers that in one glance.
   777→      </p>
   778→    </SectionCard>
   779→  );
   780→}
