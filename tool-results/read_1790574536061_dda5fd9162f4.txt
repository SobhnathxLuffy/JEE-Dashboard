    46→    46→/** E2: consecutive days where all four blocks are true — ends today if today is
    47→    47→ *  complete, otherwise yesterday (today still in progress never breaks it). */
    48→    48→function computeStreak(logs: DailyLog[], today: string): number {
    49→    49→  const byDate = new Map(logs.map((l) => [l.date, l]));
    50→    50→  const allDone = (l: DailyLog | undefined) => !!l && Object.values(l.blocks).every(Boolean);
    51→    51→  let cursor = allDone(byDate.get(today)) ? today : addDays(today, -1);
    52→    52→  let streak = 0;
    53→    53→  while (allDone(byDate.get(cursor))) {
    54→    54→    streak += 1;
    55→    55→    cursor = addDays(cursor, -1);
    56→    56→  }
    57→    57→  return streak;
    58→    58→}

   639→   639→function TodayCard({
   640→   640→  log,
   641→   641→  date,
   642→   642→  streak,
   643→   643→}: {
   644→   644→  log: DailyLog | undefined;
   645→   645→  date: string;
   646→   646→  streak: number;
   647→   647→}) {
   648→   648→  const [chapters, setChapters] = useState(log?.chapters ?? "");
   649→   649→  const [editing, setEditing] = useState(false);
   650→   650→
   651→   651→  // E5: read the FRESHEST row before writing — a block toggle must never
   652→   652→  // clobber a concurrent write or the unsaved "tonight's chapters" text.
   653→   653→  // The daily_log object store is keyed "id" (like every non-kv store), so
   654→   654→  // rows are persisted as { id: date, ...log } — a bare {date,…} row throws
   655→   655→  // DataError on put (pre-existing bug: the old Today card never persisted).
   656→   656→  async function freshRow(): Promise<DailyLog & { id: string }> {
   657→   657→    const row = await get("daily_log", date);
   658→   658→    if (row) return { ...row, id: date };
   659→   659→    return {
   660→   660→      id: date,
   661→   661→      date,
   662→   662→      blocks: { math: false, physics: false, chemistry: false, recall: false },
   663→   663→      chapters: "",
   664→   664→    };
   665→   665→  }
   666→   666→
   667→   667→  async function toggle(key: keyof DailyLog["blocks"]) {
   668→   668→    const fresh = await freshRow();
   669→   669→    const next: DailyLog & { id: string } = {
   670→   670→      ...fresh,
   671→   671→      id: date,
   672→   672→      blocks: { ...fresh.blocks, [key]: !fresh.blocks[key] },
   673→   673→      // carry unsaved chapter text along instead of wiping it
   674→   674→      chapters: chapters.trim() ? chapters : fresh.chapters,
   675→   675→    };
   676→   676→    await put("daily_log", next);
   677→   677→  }
   678→   678→
   679→   679→  async function saveChapters() {
   680→   680→    const fresh = await freshRow();
   681→   681→    const next: DailyLog & { id: string } = { ...fresh, id: date, chapters, date };
   682→   682→    await put("daily_log", next);
   683→   683→    setEditing(false);
   684→   684→  }
   685→   685→
   686→   686→  const done = log ? BLOCKS.filter((b) => log.blocks[b.key]).length : 0;
   687→   687→
   688→   688→  return (
   689→   689→    <SectionCard
   690→   690→      title="Today"
   691→   691→      subtitle={date}
   692→   692→      action={
   693→   693→        <div className="flex items-center gap-1.5">
   694→   694→          {/* E2: streak chip — plain text, subtle */}
   695→   695→          {streak >= 1 ? (
   696→   696→            <Badge
   697→   697→              variant="outline"
   698→   698→              className={
   699→   699→                streak >= 3
   700→   700→                  ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   701→   701→                  : "border-stone-300 text-stone-600"
   702→   702→              }
   703→   703→            >
   704→   704→              {streak}-day streak
   705→   705→            </Badge>
   706→   706→          ) : null}
   707→   707→          <Badge
   708→   708→            variant="outline"
   709→   709→            className={
   710→   710→              done === 4
   711→   711→                ? "border-emerald-300 bg-emerald-50 text-emerald-700"
   712→   712→                : "border-stone-300 text-stone-500"
   713→   713→            }
   714→   714→          >
   715→   715→            {done}/4 blocks
   716→   716→          </Badge>
   717→   717→        </div>
   718→   718→      }
   719→   719→    >
   720→   720→      <ul className="space-y-2">
   721→   721→        {BLOCKS.map((b) => (
   722→   722→          <li key={b.key}>
   723→   723→            <label className="flex items-center gap-3 rounded-lg border border-stone-200 px-3 py-2.5 cursor-pointer hover:bg-stone-50 transition-colors">
   724→   724→              <Checkbox
   725→   725→                checked={log?.blocks[b.key] ?? false}
   726→   726→                onCheckedChange={() => void toggle(b.key)}
   727→   727→                aria-label={b.label}
   728→   728→              />
   729→   729→              <span
   730→   730→                className={
   731→   731→                  log?.blocks[b.key]
   732→   732→                    ? "text-sm text-stone-400 line-through"
   733→   733→                    : "text-sm text-stone-800"
   734→   734→                }
   735→   735→              >
   736→   736→                {b.label}
   737→   737→              </span>
   738→   738→            </label>
   739→   739→          </li>
   740→   740→        ))}
   741→   741→      </ul>
   742→   742→      <div className="mt-3">
   743→   743→        <div className="text-[11px] uppercase tracking-wide text-stone-400 font-medium mb-1">
   744→   744→          Tonight&apos;s chapters
   745→   745→        </div>
   746→   746→        {editing ? (
   747→   747→          <div className="flex gap-2">
   748→   748→            <Input
   749→   749→              value={chapters}
   750→   750→              onChange={(e) => setChapters(e.target.value)}
   751→   751→              placeholder="e.g. Electrostatics PYQs, GOC notes, 1-3-7 rev of Kinematics"
   752→   752→              onKeyDown={(e) => {
   753→   753→                if (e.key === "Enter") void saveChapters();
   754→   754→              }}
   755→   755→              onBlur={() => void saveChapters()}
   756→   756→              autoFocus
   757→   757→            />
   758→   758→            <Button size="sm" onClick={() => void saveChapters()} className="bg-emerald-700 hover:bg-emerald-800">
   759→   759→              Save
   760→   760→            </Button>
   761→   761→          </div>
   762→   762→        ) : (
   763→   763→          <button
   764→   764→            className="w-full text-left text-sm rounded-lg border border-dashed border-stone-300 px-3 py-2.5 text-stone-600 hover:bg-stone-50 transition-colors"
   765→   765→            onClick={() => {
   766→   766→              setChapters(log?.chapters ?? "");
   767→   767→              setEditing(true);
   768→   768→            }}
   769→   769→          >
   770→   770→            {log?.chapters ? log.chapters : "Click to write what you will physically do tonight…"}
   771→   771→          </button>
   772→   772→        )}
   773→   773→      </div>
   774→   774→      <p className="text-[11px] text-stone-400 mt-3">
   775→   775→        A plan fails if Tuesday night arrives and you don&apos;t know what to physically do. This
   776→   776→        card answers that in one glance.
   777→   777→      </p>
   778→   778→    </SectionCard>
   779→   779→  );
   780→   780→}
