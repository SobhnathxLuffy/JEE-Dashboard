    64→export function ResultsView({ testId, nav }: { testId: string; nav: NavController }) {
    65→  const tests = useLive("tests");

    68→  const [subjectFilter, setSubjectFilter] = useState<SubjectFilter>("all");

    70→  // D3 "next untagged" jump state
    71→  const [highlightId, setHighlightId] = useState<string | null>(null);
    72→  const jumpIdxRef = useRef(0);
    73→  const highlightTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    74→
    75→  // D7: first-save toast bookkeeping (per response, this visit)
    76→  const notedOnceRef = useRef<Set<string>>(new Set());
    77→
    78→  const test = useMemo(() => tests.find((t) => t.id === testId), [tests, testId]);
    79→
    80→  // ── late answer-key upload: paste / key-PDF / grid → full re-score ──
    81→  const [lkOpen, setLkOpen] = useState(false);
    82→  const [lkText, setLkText] = useState("");
    83→  const [lkGrid, setLkGrid] = useState<Record<number, string>>({});
    84→  const [lkOptionNums, setLkOptionNums] = useState(false);
    85→  const [lkTol, setLkTol] = useState("0");
    86→  const [lkBusy, setLkBusy] = useState(false);
    87→  const [lkFileName, setLkFileName] = useState<string | null>(null);
    88→  const lkKeyFileRef = useRef<HTMLInputElement>(null);
    89→  const lkEditedRef = useRef<Set<number>>(new Set());
    90→
    91→  const lkParsed = useMemo(() => parseAnswerKey(lkText), [lkText]);
    92→  const lkRange = test?.pdf_meta ? paperRangeOf(test.pdf_meta) : null;
    93→  const lkNos = useMemo(() => {
    94→    if (!lkRange) return [];
    95→    const out: number[] = [];
    96→    for (let n = lkRange.first; n <= lkRange.last; n++) out.push(n);
    97→    return out;
    98→  }, [lkRange]);
    99→  const lkRenderCells = lkNos.slice(0, LK_MAX_CELLS);
   100→
   101→  // paste/key-PDF parse → bulk-fill the grid (hand-typed cells always win)
   102→  useEffect(() => {
   103→    if (lkParsed.length === 0) return;
   104→    setLkGrid((g) => {
   105→      const next = { ...g };
   106→      for (const k of lkParsed) {
   107→        if (lkEditedRef.current.has(k.no)) continue;
   108→        next[k.no] = k.bonus ? "bonus" : k.answers ? k.answers.join("/") : k.answer;
   109→      }
   110→      return next;
   111→    });
   112→  }, [lkParsed]);
   113→
   114→  const lkGridKey = useMemo(() => {
   115→    const out: PdfKeyEntry[] = [];
   116→    for (const no of lkNos) {
   117→      const raw = lkGrid[no];
   118→      if (raw === undefined) continue;
   119→      const r = parseCell(raw, lkOptionNums);
   120→      if (r.kind === "entry") out.push({ ...r.entry, no });
   121→    }
   122→    return out;
   123→  }, [lkNos, lkGrid, lkOptionNums]);
   124→
   125→  const lkMissing = useMemo(() => {
   126→    const have = new Set(lkGridKey.map((k) => k.no));
   127→    return lkNos.filter((no) => !have.has(no));
   128→  }, [lkGridKey, lkNos]);
   129→
   130→  function setLkCell(no: number, raw: string) {
   131→    lkEditedRef.current.add(no);
   132→    setLkGrid((g) => {
   133→      const next = { ...g };
   134→      if (raw.trim() === "") delete next[no];
   135→      else next[no] = raw;
   136→      return next;
   137→    });
   138→  }
   139→
   140→  function openLateKey() {
   141→    if (!test?.pdf_meta) return;
   142→    // prefill: existing key (replace flow) + stored tolerance
   143→    const g: Record<number, string> = {};
   144→    for (const k of test.pdf_meta.key) {
   145→      g[k.no] = k.bonus ? "bonus" : k.answers ? k.answers.join("/") : k.answer;
   146→    }
   147→    lkEditedRef.current = new Set();
   148→    setLkGrid(g);
   149→    setLkText("");
   150→    setLkFileName(null);
   151→    setLkOptionNums(false);
   152→    setLkTol(String(test.pdf_meta.tolerance ?? 0));
   153→    setLkOpen(true);
   154→  }
   155→
   156→  async function onLkKeyFile(e: React.ChangeEvent<HTMLInputElement>) {
   157→    const file = e.target.files?.[0];
   158→    e.target.value = "";
   159→    if (!file) return;
   160→    try {
   161→      const text = await extractKeyTextFromPdf(file);
   162→      const parsed = parseAnswerKey(text);
   163→      if (parsed.length < 2) {
   164→        toast.error("No readable text — this key looks scanned. Paste it instead.");
   165→        return;
   166→      }
   167→      setLkFileName(file.name);
   168→      setLkText(text);
   169→      toast.success(`${parsed.length} keys extracted from key PDF`);
   170→    } catch (err) {
   171→      toast.error(`Could not read key PDF: ${String(err)}`);
   172→    }
   173→  }
   174→
   175→  async function applyLkKey() {
   176→    if (!test) return;
   177→    if (lkGridKey.length === 0) {
   178→      toast.error("The key grid is empty — paste the key, upload a key PDF, or fill cells");
   179→      return;
   180→    }
   181→    if (lkMissing.length > 0) {
   182→      const ok = window.confirm(
   183→        `${lkMissing.length} of ${lkNos.length} questions have no key (${lkMissing
   184→          .slice(0, 8)
   185→          .join(", ")}${lkMissing.length > 8 ? "…" : ""}). They score 0 — attempted or not. Apply anyway?`
   186→      );
   187→      if (!ok) return;
   188→    }
   189→    setLkBusy(true);
   190→    try {
   191→      const res = await applyLateKey(test.id, lkGridKey, Number(lkTol) || 0);
   192→      setLkOpen(false);
   193→      toast.success(
   194→        `Key applied — ${res.scoredRows} answer${res.scoredRows === 1 ? "" : "s"} re-checked, score ${res.score}/${res.max}`
   195→      );
   196→    } catch (err) {
   197→      toast.error(`Could not apply the key: ${String(err)}`);
   198→    } finally {
   199→      setLkBusy(false);
   200→    }
   201→  }
   202→
   203→  // D8: shared photo file input + full-size viewer dialog
   204→  const photoInputRef = useRef<HTMLInputElement>(null);
   205→  const photoTargetRef = useRef<string | null>(null);
   206→  const [photoDragId, setPhotoDragId] = useState<string | null>(null);
   207→  const [viewPhoto, setViewPhoto] = useState<{ url: string; title: string; id: string } | null>(
   208→    null
   209→  );
   210→
   211→  const rows = useMemo(() => {
   212→    const own = responses.filter((r) => r.test_id === testId);
   213→    // review reads in paper/test order — q_no is the stored number, position is the fallback
   214→    return [...own].sort((a, b) => (a.q_no ?? 0) - (b.q_no ?? 0));
   215→  }, [responses, testId]);
   216→
   217→  // D4: key-later tests stay in self-mark mode for their whole life (score is
   218→  // recomputed to a number on the first toggle, so the flag is the discriminator)
   219→  const selfMark = test?.pdf_meta?.key_later === true;
   220→
   221→  // D3: displayed number = the paper/test number stored at submit (q_no),
   222→  // falling back to the row's stable position — never re-indexed by filters.
   223→  const qNoOf = useMemo(() => {
   224→    const m = new Map<string, number>();
   225→    rows.forEach((r, i) => m.set(r.id, r.q_no ?? i + 1));
   226→    return m;
   227→  }, [rows]);
   228→
   229→  const stats = useMemo(() => {
   230→    const attempted = rows.filter((r) => r.attempted);
   231→    const correct = rows.filter((r) => r.correct === true);
   232→    const wrong = rows.filter((r) => r.attempted && r.correct === false);
   233→    const untagged = wrong.filter((r) => !r.error_tag).length;
   234→    const time = rows.reduce((a, r) => a + (r.time_spent || 0), 0);
   235→    return {
   236→      attempted: attempted.length,
   237→      correct: correct.length,
   238→      wrong: wrong.length,
   239→      untagged,
   240→      time,
   241→      accuracy: attempted.length ? Math.round((correct.length / attempted.length) * 100) : 0,
   242→      attemptRate: rows.length ? Math.round((attempted.length / rows.length) * 100) : 0,
   243→      pending: attempted.filter((r) => r.correct === null).length,
   244→    };
   245→  }, [rows]);
   246→
   247→  const shown = useMemo(() => {
   248→    let out = rows;
   249→    if (subjectFilter !== "all") out = out.filter((r) => r.subject === subjectFilter);
   250→    if (filter === "wrong") out = out.filter((r) => r.attempted && r.correct === false);
   251→    if (filter === "untagged")
   252→      out = out.filter((r) => r.attempted && r.correct === false && !r.error_tag);
   253→    return out;
   254→  }, [rows, filter, subjectFilter]);
   255→
   256→  // D1: retry candidates — wrong + unattempted-but-keyed (key-later: attempted only)
   257→  const retryIds = useMemo(() => {
   258→    if (!test?.question_ids) return [];
   259→    const keyLater = test.pdf_meta?.key_later === true;
   260→    const candidates = rows.filter((r) => {
   261→      if (keyLater) return r.attempted;
   262→      if (r.attempted) return r.correct === false;
   263→      return r.correct_answer !== "?"; // skipped but keyed → retry it too
   264→    });
   265→    const want = new Set(candidates.map((r) => r.question_id));
   266→    return test.question_ids.filter((id) => want.has(id)); // test order preserved
   267→  }, [test, rows]);

   269→  function startRetry() {
   270→    if (!test || retryIds.length === 0) return;
   271→    const retryRows = rows.filter((r) => retryIds.includes(r.question_id));
   272→    const session: ActiveSession = {
   273→      key: "active",
   274→      test_id: uid(),
   275→      mode: "cbt",
   276→      test_type: "mixed",
   277→      name: `Retry — ${test.name}`,
   278→      subject_order: SUBJECTS.filter((s) => retryRows.some((r) => r.subject === s)),
   279→      question_ids: retryIds,
   280→      duration_min: Math.max(1, retryIds.length), // 1 min per question rule
   281→      started_at: Date.now(),
   282→      answers: {},
   283→      marked: [],
   284→      current: 0,
   285→      q_times: {},
   286→      q_entered_at: Date.now(),
   287→    };
   288→    nav.startSession(session);
   289→  }
   290→
   291→  // D3: cycle through wrong+untagged rows, scroll + highlight each in turn
   292→  const untaggedIds = useMemo(
   293→    () =>
   294→      rows
   295→        .filter((r) => r.attempted && r.correct === false && !r.error_tag)
   296→        .map((r) => r.id),
   297→    [rows]
   298→  );
   299→
   300→  function jumpToNextUntagged() {
   301→    if (untaggedIds.length === 0) return;
   302→    const idx = jumpIdxRef.current % untaggedIds.length;
   303→    jumpIdxRef.current += 1;
   304→    const id = untaggedIds[idx];
   305→    const el = document.getElementById(`resp-${id}`);
   306→    if (el) el.scrollIntoView({ behavior: "smooth", block: "center" });
   307→    setHighlightId(id);
   308→    if (highlightTimerRef.current) clearTimeout(highlightTimerRef.current);
   309→    highlightTimerRef.current = setTimeout(() => setHighlightId(null), 2500);
   310→  }
   311→
   312→  /** Fresh read + merge — the row prop may be stale when several edits race. */
   313→  async function patchResponse(id: string, patch: Partial<ResponseRecord>) {
   314→    const fresh = await get("responses", id);
   315→    if (!fresh) return null;
   316→    const next = { ...fresh, ...patch };
   317→    await put("responses", next);
   318→    return next;
   319→  }
   320→
   321→  async function tagResponse(r: ResponseRecord, tag: ErrorTag) {
   322→    const updated = await patchResponse(r.id, { error_tag: tag });
   323→    if (!updated) return;
   324→    if (tag === "F") {
   325→      const existing = await get("formula", `${updated.test_id}:${updated.question_id}`);
   326→      if (!existing) {
   327→        await put("formula", {
   328→          id: `${updated.test_id}:${updated.question_id}`,
   329→          test_id: updated.test_id,
   330→          question_id: updated.question_id,
   331→          subject: updated.subject,
   332→          chapter: updated.chapter,
   333→          snippet: updated.question_snippet,
   334→          created_at: Date.now(),
   335→        });
   336→        toast.success("Added to formula sheet (F-tag)");
   337→      }
   338→    } else {
   339→      // if retagged away from F, remove from formula sheet
   340→      const existing = await get("formula", `${updated.test_id}:${updated.question_id}`);
   341→      if (existing) {
   342→        const { del } = await import("@/lib/idb");
   343→        await del("formula", existing.id);
   344→      }
   345→    }
   346→  }
   347→
   348→  // D7: one-line note, saved on blur (Enter blurs). Toast on the first save of each note only.
   349→  async function saveNote(r: ResponseRecord, value: string) {
   350→    const v = value.trim();
   351→    if ((r.note ?? "") === v) return; // nothing changed
   352→    await patchResponse(r.id, { note: v === "" ? undefined : v });
   353→    if (!notedOnceRef.current.has(r.id)) {
   354→      notedOnceRef.current.add(r.id);
   355→      toast.success("Note saved");
   356→    }
   357→  }

   359→  // D8: photo attach/replace via the shared hidden file input
   360→  function pickPhoto(responseId: string) {

   362→    photoInputRef.current?.click();
   363→  }

   365→  async function attachPhoto(file: File, responseId: string) {
   366→    try {
   367→      const dataUrl = await fileToDataUrl(file, 1000);
   368→      const updated = await patchResponse(responseId, { photo: dataUrl });
   369→      if (updated) toast.success("Solution photo attached");
   370→    } catch {
   371→      toast.error("Could not read that image — try a JPG/PNG");

   375→  async function onPhotoFile(e: React.ChangeEvent<HTMLInputElement>) {
   376→    const file = e.target.files?.[0];

   378→    const targetId = photoTargetRef.current;
   379→    photoTargetRef.current = null;

   384→  async function removePhoto(id: string) {

   390→  // D4: set the mark, then recompute the whole TestRecord from ALL responses
   391→  async function applyMark(r: ResponseRecord, correct: boolean | null) {
   392→    if (!test) return;
   393→    const updated = await patchResponse(r.id, { correct });
   394→    if (!updated) return;
   395→    const all = (await getAll("responses")).filter((x) => x.test_id === testId);
   396→    const subjectScores: Record<Subject, number> = {
   397→      Physics: 0,
   398→      Chemistry: 0,
   399→      Mathematics: 0,
   400→    };
   401→    for (const x of all) subjectScores[x.subject] += marksFor(x.attempted, x.correct);
   402→    const score = SUBJECTS.reduce((a, s) => a + subjectScores[s], 0);
   403→    const attemptedRows = all.filter((x) => x.attempted);
   404→    const allMarked =
   405→      attemptedRows.length > 0 && attemptedRows.every((x) => x.correct !== null);
   406→    const wasAllMarked = rows
   407→      .filter((x) => x.attempted)
   408→      .every((x) => x.correct !== null);
   409→    await put("tests", { ...test, score, subject_scores: subjectScores });
   410→    if (!wasAllMarked && allMarked) {
   411→      toast.success(`Score finalised: ${score}/${test.max_score}`);
   412→    }
   413→  }
   414→
   415→  if (!test) {
   416→    return (
   417→      <div className="space-y-4">
   418→        <PageTitle title="Test analysis" />
   419→        <EmptyNote>
   420→          Test not found.{" "}
   421→          <button className="underline" onClick={() => nav.go("dashboard")}>
   422→            Back to dashboard
   423→          </button>
   424→        </EmptyNote>
   425→      </div>
   426→    );
   427→  }
   428→
   429→  const markedCount = stats.attempted - stats.pending;
   430→
   431→  return (
   432→    <div className="space-y-6">
   433→      <PageTitle
   434→        title={test.name}
   435→        subtitle={`${test.date} · ${test.source} · ${test.duration_min} min · ${test.type} test`}
   436→        right={
   437→          <Button variant="outline" size="sm" onClick={() => nav.go("dashboard")}>
   438→            ← Dashboard
   439→          </Button>
   440→        }
   441→      />
   442→
   443→      {/* score hero — ring draws in, number settles, subject split at a glance */}
   444→      <div className="rounded-xl border border-border bg-card card-shadow px-5 py-4 flex items-center gap-6 flex-wrap">
   445→        {test.score === null ? (
   446→          <div className="flex items-center gap-3 text-sm text-amber-700 dark:text-amber-300 py-2">
   447→            <span className="w-2.5 h-2.5 rounded-full bg-amber-500 animate-pulse" aria-hidden="true" />
   448→            Score pending — add the answer key or finish self-marking below.
   449→          </div>
   450→        ) : (
   451→          <>
   452→            <ScoreRing percent={(test.score / Math.max(1, test.max_score)) * 100} size={92}>
   453→              <div className="text-center leading-none">
   454→                <div className="font-mono tabular-nums text-xl font-bold text-foreground">
   455→                  <CountUp value={test.score} />
   456→                </div>
   457→                <div className="text-[10px] font-mono text-muted-foreground mt-0.5">/{test.max_score}</div>
   458→              </div>
   459→            </ScoreRing>
   460→            <div className="flex flex-wrap items-center gap-x-7 gap-y-3 min-w-0">
   461→              <div>
   462→                <div className="text-[10px] uppercase tracking-[0.08em] text-muted-foreground font-medium">Accuracy</div>
   463→                <div className="font-mono tabular-nums text-lg font-semibold text-foreground">
   464→                  <CountUp value={stats.accuracy} format={(v) => `${Math.round(v)}%`} />
   465→                </div>
   466→              </div>
   467→              <div>
   468→                <div className="text-[10px] uppercase tracking-[0.08em] text-muted-foreground font-medium">Attempted</div>
   469→                <div className="font-mono tabular-nums text-lg font-semibold text-foreground">
   470→                  {stats.attempted}
   471→                  <span className="text-muted-foreground/60 text-sm">/{rows.length}</span>
   472→                </div>
   473→              </div>
   474→              <div>
   475→                <div className="text-[10px] uppercase tracking-[0.08em] text-muted-foreground font-medium">Time</div>
   476→                <div className="font-mono tabular-nums text-lg font-semibold text-foreground">{fmtSecs(stats.time)}</div>
   477→              </div>
   478→              <div className="flex items-center gap-3 flex-wrap">
   479→                {SUBJECTS.map((s) => (
   480→                  <div key={s} className="flex items-center gap-1.5">
   481→                    <span
   482→                      className={cn(
   483→                        "inline-block w-2 h-2 rounded-full",
   484→                        s === "Physics" ? "bg-amber-500" : s === "Chemistry" ? "bg-sage-600 dark:bg-sage-400" : "bg-stone-800 dark:bg-stone-300"
   485→                      )}
   486→                      aria-hidden="true"
   487→                    />
