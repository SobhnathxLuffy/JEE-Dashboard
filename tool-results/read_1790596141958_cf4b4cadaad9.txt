   390→   390→  // D4: set the mark, then recompute the whole TestRecord from ALL responses
   391→   391→  async function applyMark(r: ResponseRecord, correct: boolean | null) {
   392→   392→    if (!test) return;
   393→   393→    const updated = await patchResponse(r.id, { correct });
   394→   394→    if (!updated) return;
   395→   395→    const all = (await getAll("responses")).filter((x) => x.test_id === testId);
   396→   396→    const subjectScores: Record<Subject, number> = {
   397→   397→      Physics: 0,
   398→   398→      Chemistry: 0,
   399→   399→      Mathematics: 0,
   400→   400→    };
   401→   401→    for (const x of all) subjectScores[x.subject] += marksFor(x.attempted, x.correct);
   402→   402→    const score = SUBJECTS.reduce((a, s) => a + subjectScores[s], 0);
   403→   403→    const attemptedRows = all.filter((x) => x.attempted);
   404→   404→    const allMarked =
   405→   405→      attemptedRows.length > 0 && attemptedRows.every((x) => x.correct !== null);
   406→   406→    const wasAllMarked = rows
   407→   407→      .filter((x) => x.attempted)
   408→   408→      .every((x) => x.correct !== null);
   409→   409→    await put("tests", { ...test, score, subject_scores: subjectScores });
   410→   410→    if (!wasAllMarked && allMarked) {
   411→   411→      toast.success(`Score finalised: ${score}/${test.max_score}`);
   412→   412→    }
   413→   413→  }
   414→   414→
   415→   415→  if (!test) {
   416→   416→    return (
   417→   417→      <div className="space-y-4">
   418→   418→        <PageTitle title="Test analysis" />
