    47→/** E3b: tooltip for the error-tags bar — raw count + share of tagged wrong. */
    48→function TagPctTooltip({
    49→  active,
    50→  payload,
    51→}: {
    52→  active?: boolean;
    53→  payload?: { payload?: { tag?: string; count?: number; pct?: number } }[];
    54→}) {
    55→  if (!active || !payload || payload.length === 0 || !payload[0]?.payload) return null;
    56→  const d = payload[0].payload;
    57→  return (
    58→    <div className="rounded-lg border border-border bg-popover text-popover-foreground px-2.5 py-1.5 text-xs shadow-[0_8px_24px_-12px_rgba(28,25,23,0.25)]">
    59→      <div className="font-medium text-foreground">{d.tag}</div>
    60→      <div className="text-muted-foreground tabular-nums">
    61→        {d.count} wrong · {d.pct}% of tagged
    62→      </div>
    63→    </div>
    64→  );
    65→}
